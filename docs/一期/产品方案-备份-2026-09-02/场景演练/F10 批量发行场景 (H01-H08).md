// 实现逻辑
async function scanDirectory(directoryPath: string): Promise<ScanResult> {
  const results = {
    totalFolders: 0,
    validProjects: [],
    invalidEntries: []
  };
  
  // 递归扫描
  const allDirs = await fs.readdirRecursive(directoryPath);
  
  for (const dir of allDirs) {
    const manifestPath = path.join(dir, 'manifest.json');
    
    if (!await fs.exists(manifestPath)) {
      results.invalidEntries.push({
        type: 'missing_manifest',
        path: dir
      });
      continue;
    }
    
    try {
      const manifest = JSON.parse(await fs.readFile(manifestPath));
      
      // 验证必需字段
      if (!validateManifestSchema(manifest)) {
        throw new Error('INVALID_SCHEMA');
      }
      
      // 识别项目类型
      const projectType = detectProjectType(manifest);
      if (projectType === 'unknown') {
        results.invalidEntries.push({
          type: 'not_theme_project',
          path: dir,
          reason: `Unknown type: ${manifest.type}`
        });
        continue;
      }
      
      // 确定操作类型
      const action = await determineAction(manifest, dir);
      
      results.validProjects.push({
        directory: dir,
        manifest,
        projectType,
        action  // 'new_publish' | 'version_update' | 'listing_update'
      });
    } catch (error) {
      results.invalidEntries.push({
        type: error.code === 'INVALID_SCHEMA' ? 'invalid_json' : 'scan_error',
        path: dir,
        reason: error.message
      });
    }
  }
  
  return results;
}

interface ScanResult {
  totalFolders: number;
  validProjects: Array<{
    directory: string;
    manifest: Manifest;
    projectType: 'theme' | 'plugin' | 'library';
    action: 'new_publish' | 'version_update' | 'listing_update';
  }>;
  invalidEntries: Array<{
    type: string;
    path: string;
    reason?: string;
  }>;
}

// 实现细节
async function checkFrozenResources(projects: ValidProject[]): Promise<FrozenCheckResult> {
  const frozenList: FrozenResource[] = [];
  const allowedList: ValidProject[] = [];
  
  for (const project of projects) {
    if (project.action === 'new_publish') {
      // 新建项目不检查冻结
      allowedList.push(project);
      continue;
    }
    
    // 已有资源，检查冻结状态
    const resource = await platform.getResourceByAssetId(project.manifest.name);
    
    if (resource.frozen === 1) {
      frozenList.push({
        resourceId: resource.id,
        resourceName: resource.name,
        frozenSince: resource.frozenAt,
        frozenReason: resource.frozenReason,
        frozenBy: resource.frozenBy
      });
    } else {
      allowedList.push(project);
    }
  }
  
  return {
    frozenList,
    allowedList,
    statistics: {
      total: projects.length,
      frozen: frozenList.length,
      allowed: allowedList.length
    }
  };
}

interface FrozenResource {
  resourceId: string;
  resourceName: string;
  frozenSince: string;
  frozenReason: string;
  frozenBy: string;
}

// 批次划分算法
function partitionProjects(
  projects: ValidProject[],
  batchSize: number,
  maxConcurrency: number
): PartitionStrategy {
  const batches = [];
  
  // 分批
  for (let i = 0; i < projects.length; i += batchSize) {
    const batchItems = projects.slice(i, i + batchSize);
    batches.push(batchItems);
  }
  
  // 创建执行队列
  const executionQueue = {
    batches,
    semaphore: new Semaphore(maxConcurrency),
    retryPolicy: {
      maxRetries: 3,
      backoff: 'exponential',
      initialDelay: 1000,
      maxDelay: 10000
    },
    betweenBatchDelay: 500  // ms
  };
  
  return executionQueue;
}

// 并发控制实现
class BatchPublisher {
  private semaphore: Semaphore;
  private queue: ItemQueue;
  
  async execute(): Promise<BatchReport> {
    const report = {
      startTime: new Date(),
      batches: [],
      items: []
    };
    
    // 分批处理
    for (const batch of this.partitionedBatches) {
      const batchStart = Date.now();
      
      // 并发执行批次内项目
      const batchResults = await Promise.allSettled(
        batch.map(item => this.processItemWithRetry(item))
      );
      
      const batchEnd = Date.now();
      
      report.batches.push({
        batchNumber: this.currentBatch++,
        items: batchResults,
        duration: batchEnd - batchStart,
        successCount: batchResults.filter(r => r.status === 'fulfilled').length,
        failCount: batchResults.filter(r => r.status === 'rejected').length
      });
      
      // 批次间隔，避免 API 限流
      await sleep(this.config.betweenBatchDelay);
      
      // 实时更新进度条
      this.progressBar.update(this.currentBatch / this.totalBatches);
    }
    
    report.endTime = new Date();
    return report;
  }
  
  private async processItemWithRetry(item: BatchItem): Promise<ItemResult> {
    let lastError: Error;
    
    for (let attempt = 1; attempt <= this.config.maxRetries; attempt++) {
      try {
        return await this.publishSingle(item);
      } catch (error) {
        lastError = error as Error;
        
        if (attempt < this.config.maxRetries) {
          const delay = calculateBackoff(attempt, this.config);
          console.log(chalk`{yellow 重试中... (#{attempt}/${this.config.maxRetries})}`);
          await sleep(delay);
        }
      }
    }
    
    return {
      status: 'failed',
      item,
      error: lastError?.message,
      retries: this.config.maxRetries
    };
  }
}

async function handleRateLimit(error: ApiError): Promise<RateLimitResponse> {
  const limitInfo = parseRateLimitHeaders(error.response.headers);
  
  // 记录到 checkpoint
  saveCheckpoint({
    step: 'rate_limit_detected',
    currentBatch: currentBatchNumber,
    failedItemCount: failedCountInBatch,
    rateLimitInfo: {
      remaining: limitInfo.remaining,
      resetInMs: limitInfo.resetInMilliseconds,
      retryAfter: limitInfo.retryAfterSeconds
    }
  });
  
  // TTY 提示
  console.log(chalk`

${limitInfo.remaining}/${limitInfo.total} requests available
Reset in: ${(limitInfo.resetInMilliseconds / 1000).toFixed(1)}s

💡 系统将自动等待后继续...
  `);
  
  // 等待限流重置

    waitingTimeMs: limitInfo.resetInMilliseconds + 500
  });
  
  return { shouldRetry: true, waitComplete: true };
}

function parseRateLimitHeaders(headers: Record<string, any>): RateLimitInfo {
  return {
    total: parseInt(headers['X-RateLimit-Limit'] || '60'),
    remaining: parseInt(headers['X-RateLimit-Remaining'] || '0'),
    resetInMilliseconds: parseInt(headers['X-RateLimit-Reset'] || '0') * 1000,
    retryAfterSeconds: parseInt(headers['Retry-After'] || '0')
  };

    for (const item of [...report.successItems, ...report.failItems, ...report.skipItems]) {
      const status = item.status.charAt(0).toUpperCase() + item.status.slice(1);
      const size = item.sizeMB.toFixed(2);
      const duration = (item.durationMs / 1000).toFixed(1);
      const errorCode = item.errorCode || '-';
      const resourceId = item.resourceId || '-';
      
      csv += `${status},${item.name},${size},${duration},${errorCode},${resourceId}\n`;
    }
    
    return csv;
  }
}

async function resumeFailedItems(checkpointId: string): Promise<ResumeResult> {
  const checkpoint = await loadCheckpoint(checkpointId);
  
  if (!checkpoint) {
    throw new UserFacingError('CHECKPOINT_MISSING', {
      hint: '无法找到 checkpoint 文件'
    });
  }
  
  // 提取失败项列表
  const failedItems = checkpoint.collectedData.failedItems || [];
  
  if (failedItems.length === 0) {
    throw new UserFacingError('NO_FAILURES_TO_RETRY', {
      hint: '没有失败项需要重试'
    });
  }
  
  console.log(chalk`\n${chalk.cyan(`🔄 准备重试 ${failedItems.length} 个失败项`)}`);
  
  // 逐个重试
  const retryResults = await Promise.allSettled(
    failedItems.map(async (item: FailedItem) => {
      try {
        // 重新执行完整的发布流程
        await publishSingleItem(item);
        
        return {
          status: 'success',
          item: item.name,
          newIndex: checkpoint.collectedData.successItems.length + retryIndex++
        };
      } catch (error) {
        return {
          status: 'failed',
          item: item.name,
          error: error.message
        };
      }
    })
  );
  
  // 生成新的 checkpoint 状态
  const newSuccesses = retryResults.filter(r => r.status === 'fulfilled');
  const newFailures = retryResults.filter(r => r.status === 'rejected');
  
  updateCheckpoint({
    step: 'retry_complete',
    newlyRecovered: newSuccesses.length,
    stillFailed: newFailures.length,
    finalStatus: newFailures.length === 0 ? 'completed' : 'partial_failure'
  });
  
  if (newFailures.length === 0) {
    deleteCheckpoint(checkpointId);  // 全部成功后删除 checkpoint
  }
  
  return {
    successfullyRetried: newSuccesses.length,
    stillFailed: newFailures.length,
    finalStatus: newFailures.length === 0 ? 'success' : 'partial_success'
  };
}
