## 📋 **一、APIClient 复用原则**

### **1. 核心规则**

```typescript
// ✅ CLI 必须使用 tools-lib 的 APIClient
import { APIClient, FreelogAPI } from '@freelog/tools-lib';

class MyCLI {
  private api: FreelogAPI;
  
  constructor(creds: CLICredentials) {
    // 初始化 API 客户端
    this.api = new APIClient({
      baseURL: 'https://api.freelog.dev',
      credentials: creds,
    });
  }
}

// ❌ 不允许在 CLI 中重新实现 API 调用逻辑
```

### **2. API 调用示例**

```typescript
// Step1: 从业务梳理中的 Step 引用 → F1.1 Step1 创建资源壳
async function createResourceShell(params: {
  type: string;
  title: string;
}) {
  // ✅ 使用 tools-lib 的 API
  const response = await api.resource.createAuthShell({
    type: params.type,
    title: params.title,
  });
  
  return response; // CreateAuthShellResponse
}
```

## 💡 **三、错误包装规范**

### **1. 统一错误处理**

```typescript
// tools-lib 返回 APIError
interface APIError {
  code: string;
  message: string;
}

// CLI 包装成 CLIError (见 05-错误码体系.md)
function wrapAPIError(error: APIError): CLIError {
  switch (error.code) {
    case 'RESOURCE_TYPE_NOT_FOUND':
      return new CLIError(CLI_ERROR_CODES.RESOURCE_TYPE_NOT_FOUND);
    case 'AUTH_ID_EXISTS':
      return new CLIError(CLI_ERROR_CODES.AUTH_ID_EXISTS, {
        authId: error.details?.authId,
      });
    default:
      return new CLIError(CLI_ERROR_CODES.API_UNKNOWN_ERROR);
  }
}
```
