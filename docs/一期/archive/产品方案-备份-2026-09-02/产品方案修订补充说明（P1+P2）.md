|--------|------|-------------|---------|------|---------|---------|

  height: number;

}

function validateCoverImage(
  imagePath: string,
  resourceType: string
): ValidationRule {
  const dimensions = getImageDimensions(imagePath);
  
  const typeRules: Record<string, string[]> = {
    "node-exhibit-theme": ["16:9", "4:3"],      // 主题推荐横图
    "plugin-documentation": ["16:9"],          // 文档推荐横图
    "mobile-app-icon": ["1:1"],                 // APP 图标必须正方形
    "video-thumbnail": ["16:9"]                 // 视频封面推荐横图
  };
  
  const allowedRatios = typeRules[resourceType] || ["any"];
  const isAcceptable = allowedRatios.includes("any") || 
                       allowedRatios.includes(dimensions.ratio);
  
  return {
    valid: true, // 平台最终接受任何尺寸
    warnings: !isAcceptable ? [
      `封面比例 ${dimensions.ratio} 不匹配该资源类型的推荐比例 ${allowedRatios.join(", ")}`
    ] : [],
    suggestion: !isAcceptable ? "建议使用 1000x563px (16:9) 作为通用封面" : null
  };
}

#### **CLI 标签推荐实现**

```typescript
class TagRecommendationService {
  private historyFile = "~/.freelog/tags.json";
  private maxSuggestions = 5;
  
  async getSuggestions(
    partialTag: string,
    resourceType: string
  ): Promise<TagSuggestion[]> {
    const suggestions: TagSuggestion[] = [];
    
    // 1. 从本地历史记录中提取
    const history = this.loadHistory();
    const matchingHistory = history.filter(t => 
      t.tag.toLowerCase().includes(partialTag.toLowerCase())
    ).slice(0, 3);
    
    suggestions.push(...matchingHistory.map(t => ({
      tag: t.tag,
      source: "history",
      usageCount: t.count
    })));
    
    // 2. 如果 partialTag 非空，调用平台热门标签 API
    if (partialTag && partialTag.length >= 2) {
      try {
        const popularTags = await platform.getPopularTags({
          resourceType,
          keyword: partialTag,
          limit: 2
        });
        
        suggestions.push(...popularTags.map(t => ({
          tag: t.name,
          source: "platform_popular",
          usageCount: t.usageCount
        })));
      } catch (error) {
        // 平台 API 失败则降级为仅显示历史标签
        console.debug("Platform tag API failed, falling back to history only");
      }
    }
    
    return suggestions.slice(0, this.maxSuggestions);
  }
  
  private loadHistory(): TagRecord[] {
    const file = fs.readFileSync(this.historyFile, "utf8");
    return JSON.parse(file).tags;
  }
}
```

#### **CLI 实现策略**

```typescript
function handleVersionDescription(
  userInput: string,
  allowHtml: boolean = false
): ProcessedDescription {
  if (allowHtml) {
    // 用户显式请求 HTML 模式
    return {
      format: "html",
      content: sanitizeHtml(userInput), // 仅保留安全标签
      rawInput: userInput
    };
  } else {
    // 默认模式：转义 HTML 字符
    return {
      format: "plaintext",
      content: escapeHtml(userInput),
      rawInput: userInput,
      detectedHtmlTags: extractHtmlTags(userInput) // 标记是否疑似 HTML
    };
  }
}

function sanitizeHtml(input: string): string {
  // 仅允许少量安全标签：<br>, <p>, <ul>, <ol>, <li>, <strong>, <em>
  const safeTags = ["br", "p", "ul", "ol", "li", "strong", "em"];
  return input.replace(/<\/?[a-z][a-z0-9]*\b[^>]*>/gi, (tag) => {
    const tagName = tag.match(/<\/?([a-z0-9]+)/i)?.[1];
    return safeTags.includes(tagName ?? "") ? tag : "";
  });
}
```

### **P1-05: 独立压缩的 ignore 规则优先级** 🔴 **待实施**

#### **背景**
独立压缩命令 `pack dir` 需要处理多层 ignore 规则的优先级冲突（全局 ignore / 项目 .gitignore / CLI 内置排除规则）。

#### **修订建议**

在 `04-CLI 流程与命令设计.md` 的独立压缩流程中增加优先级说明：

```markdown
### 3.2 独立压缩/打包流程（修订版）

```text
pack <目录>
  → 扫描目录
  → 应用 ignore 规则（按以下优先级）
      1. CLI 内置强制排除（最高优先级，无法覆盖）
      2. 命令行 --exclude 参数（用户显式指定）
      3. 项目根目录 .gitignore
      4. 项目根目录 .freelogignore（CLI 专用）
      5. 用户全局配置 ~/.freelog/config.json 中的 defaultIgnore
  → 展示将包含/忽略/拒绝的文件摘要
  → 用户确认输出路径
  → 生成确定性压缩包
  → 输出 artifact 路径、大小、sha1、文件清单和报告
```

**优先级矩阵**：

| 规则来源 | 优先级 | 可覆盖性 | 示例 |
|---------|-------|---------|------|
| CLI 内置强制排除 | P1 (最高) | ❌ 不可覆盖 | node_modules/, *.log, dist/ |
| 命令行 --exclude | P2 | ⚠️ 可被 P1 覆盖 | `--exclude secret.key` |
| 项目 .gitignore | P3 | ⚠️ 可被 P1/P2 覆盖 | `.env`, `*.tmp` |
| 项目 .freelogignore | P4 | ⚠️ 可被 P1-P3 覆盖 | 构建产物临时文件 |
| 用户全局配置 | P5 (最低) | ❌ 可被 P1-P4 全部覆盖 | 默认排除系统日志目录 |
```

#### **CLI 实现逻辑**

```typescript
function buildIgnoreList(options: PackOptions): IgnoreMatcher {
  const rules: IgnoreRule[] = [];
  
  // P1: CLI 内置强制排除（不可覆盖）
  rules.push({
    pattern: ["node_modules/", "*.log", "*.tmp", "dist/", ".git/"],
    priority: 1,
    overrideable: false
  });
  
  // P2: 命令行 --exclude 参数
  if (options.exclude) {
    options.exclude.forEach(pattern => {
      rules.push({
        pattern: [pattern],
        priority: 2,
        overrideable: true
      });
    });
  }
  
  // P3: 项目 .gitignore
  const gitignorePath = findFileInParentDirs(".gitignore", options.cwd);
  if (gitignorePath) {
    rules.push({
      pattern: parseGitignore(gitignorePath),
      priority: 3,
      overrideable: true
    });
  }
  
  // P4: 项目 .freelogignore
  const freelogignorePath = path.join(options.cwd, ".freelogignore");
  if (fs.existsSync(freelogignorePath)) {
    rules.push({
      pattern: parseSimpleIgnore(freelogignorePath),
      priority: 4,
      overrideable: true
    });
  }
  
  // P5: 用户全局配置
  const userConfig = loadUserConfig();
  if (userConfig.defaultIgnore) {
    rules.push({
      pattern: userConfig.defaultIgnore,
      priority: 5,
      overrideable: true
    });
  }
  
  // 按优先级排序，高优先级规则优先匹配
  rules.sort((a, b) => a.priority - b.priority);
  
  return createIgnoreMatcher(rules);
}
```

function shouldSkipGuid(guid: string, newFeedItem: FeedItem): boolean {
  // 1. 精确匹配 GUID
  if (existingGuidSet.has(guid)) return true;
  
  // 2. 模糊匹配：相同标题 + 发布时间差 < 1 小时
  const similarItems = existingItems.filter(item => 
    item.title === newFeedItem.title &&
    Math.abs(parseDate(item.pubDate) - parseDate(newFeedItem.pubDate)) < 3600000
  );
  
  if (similarItems.length > 0) {

    return true; // 跳过但不报错
  }
  
  return false;
}
