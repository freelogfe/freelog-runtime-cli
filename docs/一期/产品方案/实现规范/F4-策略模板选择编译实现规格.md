```text

  │
  ├─ 选项 B: 搜索关键词
  │   ├─ 支持模糊匹配 (名称/说明)
  │   ├─ 实时过滤列表
  │   └─ 从过滤结果中选择
  │
  └─ 选项 C: 返回上一步
      └─ 重新浏览其他分类
  ↓
[Step 4] 动态表单渲染
  ├─ 读取 template.schema.parameters
  ├─ 为每个参数生成表单字段
  │   ├─ fieldId、label、type(string/number/boolean/array)
  │   ├─ required(必填标记)
  │   ├─ defaultValue(默认值)
  │   ├─ validation(minLength/maxLength/pattern/enumValues)
  │   └─ hint(提示信息)
  │
  └─ TTY 逐项输入或跳过
  ↓
[Step 5] 参数校验
  ├─ 本地校验必填项
  ├─ 类型转换与格式验证
  ├─ 枚举值检查
  └─ 失败则显示具体原因并要求重新输入
  ↓
[Step 6] 调用平台编译
  ├─ 调用 platform.compilePolicy(templateId, userArgs)
  ├─ 接收编译后的 JSON 策略
  ├─ 验证 JSON Schema
  └─ 失败则显示错误路径和修复建议
  ↓
[Step 7] 策略翻译预览
  ├─ 调用 platform.translatePolicy(compiledJson, locale='zh-CN')
  ├─ 展示人类可读摘要
  │   ├─ 策略名称
  │   ├─ 核心规则摘要
  │   └─ 关键参数列表
  │
  └─ 用户确认或回到 Step 4 修改
  ↓
[Step 8] 查重检测
  ├─ 计算策略 Hash(SHA-256 of compiledJson)
  ├─ 查询平台：相同 hash 是否已存在？
  ├─ 如果重复 → 显示已有策略信息

  │   ├─ 询问用户："是否复用已有策略？"
  │   ├─ 选项 A: 复用 (直接使用 existingPolicyId)

  │   └─ 选项 C: 修改参数避免重复

  ↓

  ├─ 调用 platform.updateResource({
  │   resourceId,
  │   policyId
  │ })
  ├─ 验证返回结果
  └─ 策略绑定完成
  ↓
完成
```

#### **Step 1: 模板缓存策略**

```typescript
interface TemplateCacheStrategy {
  // 缓存 TTL:1 小时
  ttlMs: 3600 * 1000;
  
  // 缓存键生成规则
  generateKey: (resourceTypeCode: string): string => {
    return `templates:${resourceTypeCode}`;
  };
  
  // 加载逻辑 (含缓存判断)
  load: async (resourceTypeCode: string): Promise<TemplateList> => {
    const key = this.generateKey(resourceTypeCode);
    
    // 检查缓存
    const cached = await cache.get(key);
    if (cached && Date.now() - cached.timestamp < this.ttlMs) {
      console.log(chalk`{dim ℹ️} 使用缓存的模板列表 (${Math.round((this.ttlMs - (Date.now() - cached.timestamp)) / 1000)}s 后过期)`);
      return cached.data;
    }
    
    // 拉取最新数据
    console.log(chalk`{dim ℹ️} 正在拉取策略模板列表...`);
    const templates = await platform.getPolicyTemplates(resourceTypeCode);
    
    // 更新缓存
    await cache.set(key, {
      data: templates,
      timestamp: Date.now()
    });
    
    return templates;
  };
  
  // 缓存刷新命令
  forceRefresh: async (resourceTypeCode: string): Promise<TemplateList> => {
    const key = this.generateKey(resourceTypeCode);
    await cache.delete(key);  // 删除缓存
    
    return await this.load(resourceTypeCode);  // 强制重新拉取
  };
}
```

```typescript
interface TemplateListRenderer {
  // 按分类分组
  groupByCategory: (templates: TemplateListItem[]): Record<string, TemplateListItem[]> => {

    for (const template of templates) {
      const category = template.category || '其他';
      
      if (!groups[category]) {
        groups[category] = [];
      }
      
      groups[category].push(template);
    }
    
    return groups;
  };
  
  // 渲染 TTY 列表
  render: (templates: TemplateListItem[], selectedId?: string): void => {
    const groups = this.groupByCategory(templates);

    console.log(chalk`${chalk.blue('═'.repeat(70))}\n`);
    
    let index = 1;
    
    for (const [category, templatesInGroup] of Object.entries(groups)) {
      console.log(chalk`{bold yellow ${category}}\n`);
      
      for (const template of templatesInGroup) {
        const isSelected = template.templateId === selectedId;
        const marker = isSelected ? chalk.green('✓') : chalk.dim(' ');
        const badge = template.recommended ? chalk.yellowBright('[推荐]') : '';
        
        console.log(chalk`  {dim ${marker}} #{index}. {bold ${template.name}} ${badge}`);
        console.log(`     ${template.description}`);
        console.log(chalk`     {dim 适用类型：${template.resourceTypes.join(', ')}}`);
        console.log(chalk`     {dim 使用次数：${template.usageCount}}\n`);
        
        index++;
      }
      
      console.log();
    }
    
    console.log(chalk`${chalk.blue('═'.repeat(70))}`);
    console.log(`\n{italic 请输入模板编号进行选择 (或输入 R 刷新列表):}`);
  };
  
  // 搜索过滤
  filterByKeyword: (templates: TemplateListItem[], keyword: string): TemplateListItem[] => {
    if (!keyword.trim()) {
      return templates;
    }
    
    const lowerKeyword = keyword.toLowerCase();
    
    return templates.filter(template => 
      template.name.toLowerCase().includes(lowerKeyword) ||
      template.description.toLowerCase().includes(lowerKeyword)
    );
  };
  
  // 搜索 UI
  renderSearch: (templates: TemplateListItem[], keyword: string): void => {
    console.log(chalk`{dim ℹ️ 搜索结果 (关键词："${keyword}")}:`);
    this.render(templates);
  };
}
```

#### **Step 4: 动态表单渲染**

```typescript
interface DynamicFormRenderer {
  // 渲染策略参数表单
  renderForm: (parameters: ParameterDefinition[], userInputs?: Record<string, any>): FormField[] => {
    return parameters.map(param => {
      const currentValue = userInputs?.[param.name] ?? param.defaultValue;
      
      const formField: FormField = {
        fieldId: param.name,
        label: param.displayName || param.name,
        type: this.convertType(param.type),
        required: param.required || false,
        defaultValue: currentValue,
        placeholder: param.placeholder || `请输入${param.displayName}`,
        validation: {
          minLength: param.minLength,
          maxLength: param.maxLength,
          pattern: param.pattern ? new RegExp(param.pattern) : null,
          enumValues: param.enumValues,
          minValue: param.minValue,
          maxValue: param.maxValue
        },
        hint: param.description,
        options: param.enumValues  // 下拉选项
      };
      
      return formField;
    });
  };
  
  // 类型转换
  convertType: (apiType: string): FieldType => {
    const typeMap: Record<string, FieldType> = {
      'string': 'text',
      'number': 'number',
      'integer': 'number',
      'boolean': 'checkbox',
      'array': 'multiselect'
    };
    
    return typeMap[apiType] || 'text';
  };
  
  // TTY 逐项输入
  collectInputs: async (formFields: FormField[]): Promise<Record<string, any>> => {
    const inputs: Record<string, any> = {};
    
    for (const field of formFields) {
      console.log(chalk`\n{bold ${field.label}} ${field.required ? chalk.red('*') : ''}`);
      
      if (field.hint) {
        console.log(chalk`{dim ℹ️} ${field.hint}`);
      }
      
      // 显示默认值
      if (field.defaultValue !== undefined) {
        console.log(chalk`{dim 默认值：${field.defaultValue}}`);
      }
      
      let inputValue: string;
      
      switch (field.type) {
        case 'text':
        case 'number':
          inputValue = await promptInput(`请输入 [回车使用默认]: `, {
            defaultValue: field.defaultValue?.toString(),
            validate: (value) => this.validate(field, value)
          });
          break;
          
        case 'checkbox':
          inputValue = await promptConfirm(`是否启用 ${field.label}?`, {
            default: field.defaultValue || false
          });
          break;
          
        case 'multiselect':
          inputValue = await promptMultiselect(`请选择 ${field.label}`, {
            options: field.options || [],
            maxItems: 10
          });
          break;
          
        default:
          throw new Error(`不支持的字段类型：${field.type}`);
      }
      
      // 处理空值
      if (inputValue === '' && field.defaultValue !== undefined) {
        inputs[field.fieldId] = field.defaultValue;
      } else if (inputValue !== '') {
        // 类型转换
        inputs[field.fieldId] = this.parseValue(inputValue, field.type);
      } else if (!field.required) {
        inputs[field.fieldId] = null;
      } else {
        throw new UserFacingError('REQUIRED_FIELD_MISSING', {
          field: field.label
        });
      }
    }
    
    return inputs;
  };
  
  // 字段验证
  validate: (field: FormField, value: string): ValidationResult => {
    if (!value.trim() && !field.required) {
      return { valid: true };
    }
    
    const numValue = parseFloat(value);
    
    // 数字范围检查
    if (field.validation.minValue !== undefined && numValue < field.validation.minValue) {
      return {
        valid: false,
        error: `值不能小于 ${field.validation.minValue}`
      };
    }
    
    if (field.validation.maxValue !== undefined && numValue > field.validation.maxValue) {
      return {
        valid: false,
        error: `值不能大于 ${field.validation.maxValue}`
      };
    }
    
    // 长度检查
    if (field.validation.minLength !== undefined && value.length < field.validation.minLength) {
      return {
        valid: false,
        error: `长度不能少于 ${field.validation.minLength} 字符`
      };
    }
    
    if (field.validation.maxLength !== undefined && value.length > field.validation.maxLength) {
      return {
        valid: false,
        error: `长度不能超过 ${field.validation.maxLength} 字符`
      };
    }
    
    // 正则匹配
    if (field.validation.pattern && !field.validation.pattern.test(value)) {
      return {
        valid: false,
        error: `格式不符合要求`
      };
    }
    
    // 枚举值检查
    if (field.validation.enumValues && !field.validation.enumValues.includes(value)) {
      return {
        valid: false,
        error: `必须从以下选项中选择：${field.validation.enumValues.join(', ')}`
      };
    }
    
    return { valid: true };
  };
  
  // 值解析
  parseValue: (value: string, type: FieldType): any => {
    switch (type) {
      case 'number':
        return parseFloat(value);
      case 'checkbox':
        return value === 'true' || value === 'yes' || value === 'Y';
      default:
        return value;
    }
  };
}

interface FormField {
  fieldId: string;
  label: string;
  type: FieldType;
  required: boolean;
  defaultValue?: any;
  placeholder?: string;
  validation?: ValidationRule;
  hint?: string;
  options?: string[];
}

interface ParameterDefinition {
  name: string;
  displayName?: string;
  type: string;
  required?: boolean;
  description?: string;
  defaultValue?: any;
  placeholder?: string;
  minLength?: number;
  maxLength?: number;
  pattern?: string;
  enumValues?: string[];
  minValue?: number;
  maxValue?: number;
}

type FieldType = 'text' | 'number' | 'checkbox' | 'multiselect';

interface ValidationRule {
  minLength?: number;
  maxLength?: number;
  pattern?: RegExp;
  enumValues?: string[];
  minValue?: number;
  maxValue?: number;
}
```

```typescript
interface PolicyCompiler {
  // 调用平台编译
  compile: async (
    templateId: string,
    userArgs: Record<string, any>
  ): Promise<CompiledPolicy> => {
    try {
      const compiled = await platform.compilePolicy(templateId, userArgs);
      
      // JSON Schema 验证
      const isValid = validateSchema(compiled, policySchema);
      
      if (!isValid) {
        throw new UserFacingError('POLICY_SCHEMA_INVALID', {
          message: '编译后的策略不符合平台规范'
        });
      }
      
      return compiled;
    } catch (error) {
      if (error instanceof JsonSchemaError) {
        throw new UserFacingError('POLICY_COMPILE_FAILED', {
          message: '策略编译失败',
          path: error.path,
          reason: error.message,
          suggestion: `请修正 ${error.path} 字段的值`
        });
      }
      
      throw error;
    }
  };
  
  // 编译错误映射
  mapCompileError: (platformError: any): UserFacingError => {
    const errorMap: Record<string, UserFacingError> = {
      MISSING_PARAMETER: new UserFacingError('MISSING_REQUIRED_PARAM', {
        message: '缺少必填参数'
      }),
      INVALID_TYPE: new UserFacingError('PARAM_TYPE_MISMATCH', {
        message: '参数类型不正确'
      }),
      RANGE_ERROR: new UserFacingError('PARAM_OUT_OF_RANGE', {
        message: '参数值超出允许范围'
      })
    };
    
    return errorMap[platformError.code] || platformError;
  };
}

interface CompiledPolicy {
  templateId: string;
  parameters: Record<string, any>;
  hash?: string;  // 可选，由平台计算
  createdAt?: string;
}
```

```typescript
interface PolicyTranslator {
  // 翻译策略为人类可读摘要
  translate: async (compiledPolicy: CompiledPolicy, locale: string = 'zh-CN'): Promise<PrettySummary> => {
    const summary = await platform.translatePolicy(compiledPolicy, locale);
    
    return {
      templateName: summary.templateName,
      coreRules: this.extractCoreRules(summary.json),
      parameterSummary: this.summarizeParameters(summary.json.parameters),
      warnings: summary.warnings || []
    };
  };
  
  // 提取核心规则
  extractCoreRules: (policyJson: any): string[] => {
    const rules: string[] = [];
    
    // 遍历所有条件
    if (policyJson.conditions) {
      for (const condition of policyJson.conditions) {
        const ruleText = this.formatCondition(condition);
        rules.push(ruleText);
      }
    }
    
    // 遍历所有动作
    if (policyJson.actions) {
      for (const action of policyJson.actions) {
        const actionText = this.formatAction(action);
        rules.push(actionText);
      }
    }
    
    return rules;
  };
  
  // 格式化单个条件
  formatCondition: (condition: Condition): string => {
    const { field, operator, value } = condition;
    
    const fieldLabels: Record<string, string> = {
      'resource_title': '资源标题',
      'auth_identity': '授权标识',
      'resource_type': '资源类型',
      'publish_time': '发布时间',
      'download_count': '下载量'
    };
    
    const operatorLabels: Record<string, string> = {
      '=': '等于',
      '!=': '不等于',
      '>': '大于',
      '<': '小于',
      '>=': '大于等于',
      '<=': '小于等于',
      'contains': '包含',
      'starts_with': '开头是',
      'ends_with': '结尾是'
    };
    
    const fieldName = fieldLabels[field] || field;
    const operatorName = operatorLabels[operator] || operator;
    
    return `${fieldName} ${operatorName} "${value}"`;
  };
  
  // 格式化单个动作
  formatAction: (action: Action): string => {
    const { type, target } = action;
    
    const actionLabels: Record<string, string> = {
      'allow': '允许',
      'deny': '拒绝',
      'require_payment': '需付费',
      'limit_download': '限制下载',
      'require_attribution': '需署名'
    };
    
    const actionName = actionLabels[type] || type;
    
    return `${actionName} → ${target}`;
  };
  
  // 参数摘要
  summarizeParameters: (parameters: Record<string, any>): string[] => {
    const summary: string[] = [];
    
    for (const [key, value] of Object.entries(parameters)) {
      if (typeof value === 'object') {
        summary.push(`${key}: {...}`);
      } else {
        summary.push(`${key}: ${value}`);
      }
    }
    
    return summary.slice(0, 5);  // 只显示前 5 个关键参数
  };
  
  // TTY 展示摘要
  showSummary: (prettySummary: PrettySummary): void => {
    console.log(chalk`\n${chalk.blue('═'.repeat(60))}`);
    console.log(chalk`  {bold cyan 📜  策略摘要预览}`);
    console.log(chalk`${chalk.blue('═'.repeat(60))}\n`);
    
    console.log(chalk`{bold 模板名称：${prettySummary.templateName}}\n`);
    
    console.log(chalk`{bold 核心规则}:`);
    prettySummary.coreRules.forEach((rule, index) => {
      console.log(chalk`  {dim #{index + 1}.} ${rule}`);
    });
    
    console.log(chalk`\n{bold 关键参数}:`);
    prettySummary.parameterSummary.forEach((param, index) => {
      console.log(chalk`  {dim #{index + 1}.} ${param}`);
    });
    
    if (prettySummary.warnings.length > 0) {
      console.log(chalk`\n{yellow ⚠️  警告}:`);
      prettySummary.warnings.forEach(warning => {
        console.log(chalk`  {dim •} ${warning}`);
      });
    }
    
    console.log(chalk`\n{italic 确认上述策略？[Y/n]: Y}`);
    console.log(chalk`${chalk.blue('═'.repeat(60))}`);
  };
}

interface PrettySummary {
  templateName: string;
  coreRules: string[];
  parameterSummary: string[];
  warnings: string[];
}
```

#### **Step 8: 查重检测**

```typescript
interface PolicyDeduplicator {
  // 计算策略 Hash(SHA-256)
  computeHash: (policyJson: CompiledPolicy): string => {
    const normalizedJson = JSON.stringify(policyJson, Object.keys(policyJson).sort());
    const buffer = Buffer.from(normalizedJson);
    return crypto.createHash('sha256').update(buffer).digest('hex');
  };
  
  // 查询平台查重
  checkDuplicate: async (hash: string, accountId: string): Promise<DuplicateInfo | null> => {
    const duplicates = await platform.searchPoliciesByHash({
      hash,
      accountId  // 排除自身
    });
    
    if (duplicates.length > 0) {
      return {
        isDuplicate: true,
        policies: duplicates.map(p => ({
          policyId: p.policyId,
          resourceName: p.resourceName,
          createdAt: p.createdAt,
          usageCount: p.usageCount
        }))
      };
    }
    
    return null;
  };
  
  // 展示冲突信息
  showConflict: (info: DuplicateInfo): void => {
    console.log(chalk`\n{yellow ⚠️  检测到重复的策略}`);
    console.log(`\n以下资源已使用相同的策略:\n`);
    
    info.policies.forEach((policy, index) => {
      console.log(chalk`  {dim #{index + 1}.} ${policy.resourceName}`);
      console.log(chalk`     策略 ID: {cyan ${policy.policyId}}`);
      console.log(chalk`     创建时间：${policy.createdAt}`);
      console.log(chalk`     使用次数：${policy.usageCount}\n`);
    });
    
    console.log(chalk`  {italic 请决定处理方式:}`);
    console.log(`    1) 复用已有策略 (节省空间，保持一致性)`);
    console.log(`    2) 强制新建 (可能产生冗余)`);
    console.log(`    3) 修改参数避免重复`);
  };
  
  // 处理选择
  handleUserChoice: (choice: 'reuse' | 'force_new' | 'modify'): HandleResult => {
    switch (choice) {
      case 'reuse':
        return {
          action: 'USE_EXISTING',
          requiresAdditionalSteps: false,
          note: '将自动使用已有的 policyId'
        };
      
      case 'force_new':
        return {
          action: 'FORCE_NEW',
          requiresAdditionalSteps: false,
          note: '创建新策略，可能导致重复'
        };
      
      case 'modify':
        return {
          action: 'MODIFY_AND_RETRY',
          requiresAdditionalSteps: true,
          note: '返回参数填写界面修改策略参数'
        };
    }
  };
}

interface DuplicateInfo {
  isDuplicate: boolean;
  policies: Array<{
    policyId: string;
    resourceName: string;
    createdAt: string;
    usageCount: number;
  }>;
}
```

### **1.3 异常恢复机制**

```typescript
interface PolicyCheckpointHandler {
  // 保存时机
  savePoints = [
    'after_template_selected',     // 模板选中后立即保存
    'after_parameters_collected',  // 参数填写完成后
    'after_policy_compiled',       // 编译成功后
    'before_resource_update'       // 关联到资源前强制保存
  ];
  
  // checkpoint 数据结构
  structure = {
    selectedTemplateId: string;
    collectedParameters: Record<string, any>;
    compiledPolicy?: CompiledPolicy;
    policyHash?: string;
    duplicateCheckResult?: DuplicateInfo | null;
    processingTime: number;
  };
  
  // 恢复逻辑
  resume: (checkpoint: PolicyCheckpoint): ResumeDecision => {
    // 如果已编译成功且无重复，可直接关联到资源
    if (checkpoint.compiledPolicy && !checkpoint.duplicateCheckResult?.isDuplicate) {
      return {
        allowed: true,
        skipSteps: ['template_selection', 'form_collection', 'compilation'],
        preservedData: {
          templateId: checkpoint.selectedTemplateId,
          parameters: checkpoint.collectedParameters,
          compiledPolicy: checkpoint.compiledPolicy
        }
      };
    }
    
    // 已收集参数但未编译，可跳过表单直接编译
    if (checkpoint.collectedParameters) {
      return {
        allowed: true,
        skipSteps: ['template_selection', 'form_collection'],
        requiredSteps: ['compilation', 'deduplication'],
        restoredData: {
          templateId: checkpoint.selectedTemplateId,
          parameters: checkpoint.collectedParameters
        }
      };
    }
    
    // 仅选了模板，需重新填写参数
    if (checkpoint.selectedTemplateId) {
      return {
        allowed: true,
        skipSteps: ['template_selection'],
        requiredSteps: ['form_collection'],
        restoredData: {
          templateId: checkpoint.selectedTemplateId
        }
      };
    }
    
    // 无任何进度，重新开始
    return {
      allowed: false,
      reason: 'NO_PROGRESS',
      message: '无有效进度，需重新选择模板'
    };
  };
}
```
