## 📝 **二、核心交互组件**

### **2.1 合集信息表单**

```typescript
class CollectionInfoForm {
  render(currentInfo: CollectionMetadata) {
    console.log('\n┌─ 合集信息更新 ──────────────────────┐');
    console.log('│                                      │');
    
    // 标题输入
    this.inputField('标题 *', {
      value: currentInfo.title,
      maxLength: 200,
      counter: true,
      validate: (val: string) => {
        if (val.length === 0) return '标题不能为空';
        if (val.length > 200) return '标题不能超过 200 字符';
        return null;
      }
    });
    
    // 描述文本域
    this.textAreaField('描述 *', {
      value: currentInfo.description || '',
      minLength: 50,
      maxLength: 1000,
      counter: true,
      validate: (val: string) => {
        if (val.length < 50) return `描述至少需要${50}个字符`;
        if (val.length > 1000) return `描述不能超过${1000}个字符`;
        return null;
      }
    });
    
    // 封面选择
    this.imageChoiceField('封面图片', {
      currentImage: currentInfo.coverImage,
      options: [
        { value: 'keep', label: '保持现有封面', default: true },
        { value: 'upload-new', label: '重新上传封面' }
      ]
    });
    
    // 标签输入
    this.tagInput('标签', {
      tags: currentInfo.tags.map(t => t.name),
      maxTags: 10,
      autoNormalize: true
    });
    
    // 网站 URL
    this.urlField('网站 URL', {
      value: currentInfo.website || '',
      placeholder: 'https://example.com',
      optional: true
    });
    
    // 联系邮箱
    this.emailField('联系邮箱', {
      value: currentInfo.contactEmail || '',
      placeholder: 'contact@example.com',
      optional: true
    });
    
    console.log('│\n│ [提交更新] ENTER | [取消] ESC       │');
    console.log('└──────────────────────────────────────┘\n');
  }
}
```

### **2.2 封面验证器**

```typescript
class CollectionCoverValidator {
  async validate(file: File): Promise<ValidatedImage> {
    // 格式验证
    if (!ALLOWED_FORMATS.includes(file.type)) {
      throw new ValidationError('COVER_FORMAT_INVALID', 
        '仅支持 JPG/PNG/WebP 格式');
    }
    
    // 大小验证
    if (file.size > MAX_SIZE) {
      throw new ValidationError('COVER_SIZE_EXCEEDED',
        `封面图片不能大于${MAX_SIZE / 1024 / 1024}MB`);
    }
    
    // 尺寸验证 - 合集封面推荐 1200x628px
    const dimensions = await getImageDimensions(file);
    const minWidth = 800;
    const minHeight = 400;
    const recommendedWidth = 1200;
    const recommendedHeight = 628;
    
    if (dimensions.width < minWidth || dimensions.height < minHeight) {
      throw new ValidationError('COVER_DIMENSIONS_INSUFFICIENT',
        `封面最小尺寸为${minWidth}x${minHeight}, 推荐 1200x628px`);
    }
    
    // 质量检查
    const qualityScore = await analyzeImageQuality(file);
    if (qualityScore < MIN_QUALITY_THRESHOLD) {
      throw new ValidationError('COVER_QUALITY_LOW',
        '封面图片质量过低，请上传更清晰的图片');
    }
    
    return {
      file,
      sha256: await calculateSHA256(file.path),
      dimensions,
      qualityScore
    };
  }
}
```

## 🚨 **三、异常分支处理**

### **3.1 错误码清单**

| 错误码 | 触发条件 | 用户提示 | 修复建议 |
|-------|---------|---------|---------|
| COLLECTION_NOT_FOUND | 合集不存在 | "指定的合集不存在" | 检查合集 ID |
| METADATA_UPDATE_FAILED | API 调用失败 | "信息更新失败" | 检查网络后重试 |
| TITLE_TOO_LONG | 标题超长 | "标题长度为{actual}/200，请缩短" | 精简标题 |
| DESCRIPTION_TOO_SHORT | 描述过短 | "描述至少需要 50 个字符" | 补充详细说明 |
| COVER_UPLOAD_FAILED | 封面上传失败 | "封面图片上传失败" | 检查格式/大小后重传 |
| RSS_CONFLICT_INFO | RSS 锁定信息与手动修改冲突 | "部分字段已被 RSS 锁定" | 修改 RSS 配置或禁用绑定 |

### **3.2 RSS 锁定字段检测**

```typescript
function detectRSSLockedFields(
  userUpdates: Partial<CollectionMetadata>,
  rssConfig: RSSConfig | null
): LockConflict[] {
  
  if (!rssConfig) {
    return [];
  }
  
  const conflicts: LockConflict[] = [];
  
  // RSS 锁定的字段
  const lockedFields = ['title', 'description', 'website'];
  
  lockedFields.forEach(field => {
    if (userUpdates[field as keyof typeof userUpdates] !== undefined) {
      conflicts.push({
        field: field,
        reason: 'FIELD_LOCKED_BY_RSS',
        message: `"${field}"字段已被 RSS 自动同步锁定，无法手动修改`,
        suggestion: '请先解除 RSS 绑定或使用 RSS 更新'
      });
    }
  });
  
  return conflicts;
}
```

## 📤 **四、API 调用逻辑**

### **4.1 更新提交**

```typescript
async function submitCollectionInfoUpdate(
  collectionId: string,
  updates: MetadataUpdateRequest
): Promise<SubmitResult> {
  
  // POST /api/collection/metadata/update
  const response = await api.collection.updateMetadata({
    collectionId,
    updates: {
      title: updates.title ? {
        value: updates.title,
        validated: true
      } : undefined,
      
      description: updates.description ? {
        value: updates.description,
        validated: true
      } : undefined,
      
      coverImage: updates.newCoverImage ? {
        uploadId: updates.newCoverImage.uploadId,
        sha256: updates.newCoverImage.sha256,
        replace: true
      } : null, // null 表示不改变
      
      website: updates.website ? {
        value: updates.website,
        validated: true
      } : undefined,
      
      contactEmail: updates.contactEmail ? {
        value: updates.contactEmail,
        validated: true
      } : undefined
    }
  });
  
  return {
    success: true,
    updatedFields: Object.keys(response.updates).filter(k => 
      response.updates[k].hasChanged
    ),
    syncStatus: response.syncStatus
  };
}
```
