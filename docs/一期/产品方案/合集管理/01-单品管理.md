## 🔍 **二、核心功能实现**

### **2.1 条目列表渲染**

```typescript
class CollectionItemRenderer {
  render(items: CollectionItem[], rssBound?: RSSConfig) {
    console.log('\n┌─ 合集条目列表 ──────────────────────┐');
    console.log('│                                      │');
    
    items.forEach((item, index) => {
      const statusIcon = this.getStatusIcon(item.status);
      
      // 序号
      const numberColor = index === this.selectedIndex ? ui.color.primary : ui.color.success;
      console.log(`│ ${numberColor}[${String(index + 1).padStart(2)}]${ui.reset} `);
      
      // 标题和资源 ID
      console.log(`  ${item.title}`);
      console.log(`    资源 ID: ${item.resourceId}`);
      console.log(`    GUID: ${item.guid}`);
      
      // 状态
      console.log(`    状态：${statusIcon} ${this.formatStatus(item.status)}`);
      
      // 如果是 RSS 绑定且有变化
      if (rssBound && item.isRSSDerived) {
        console.log(`    ℹ️  RSS 自动收录`);
      }
      
      console.log('│');
    });
    
    console.log('│');
    console.log(`│ 总数：${items.length} 个条目              │`);
    console.log('│');
    console.log('│ [添加] A | [排序] O | [移除] R | [保存] S│');
    console.log('└──────────────────────────────────────┘\n');
  }
  
  private getStatusIcon(status: ItemStatus): string {
    switch (status) {
      case 'online': return '✅';
      case 'offline': return '❌';
      case 'unlisting': return '⚠️';
      case 'pending_sync': return '🔄';
      default: return '●';
    }
  }
}
```

### **2.2 添加条目逻辑**

```typescript
async function addItemToCollection(
  collectionId: string,
  addItemMode: 'existing' | 'new'
): Promise<AddItemResult> {
  
  if (addItemMode === 'existing') {
    // 从已有资源中选择
    const selectedResources = await promptResourceSelection();
    
    // 验证是否已在合集中
    const existingInCollection = selectedResources.filter(r => 
      await isResourceInCollection(collectionId, r.id)
    );
    
    if (existingInCollection.length > 0) {
      warn(`以下资源已存在于合集中:\n${existingInCollection.map(r => r.title).join(', ')}`);
    }
    
    const newItems = selectedResources.filter(r => 
      !existingInCollection.some(ex => ex.id === r.id)
    );
    
    // 验证 GUID 唯一性
    const duplicates = findDuplicateGuids(newItems);
    if (duplicates.length > 0) {
      throw new ValidationError('DUPLICATE_GUID', `发现重复 GUID: ${duplicates.map(d => d.guid).join(', ')}`);
    }
    
    return {
      mode: 'existing',
      addedItems: newItems.map(r => ({
        resourceId: r.id,
        guid: r.guid,
        title: r.title,
        sortOrder: this.calculateNextSortOrder()
      }))
    };
    
  } else {
    // 从本地目录创建
    const directoryPath = await promptDirectoryInput();
    
    const scanResult = await scanLocalDirectory(directoryPath);
    
    if (scanResult.episodes.length === 0) {
      throw new ValidationError('NO_EPISODES_FOUND', '目录下没有找到有效的 episode');
    }
    
    // 调用 F1 流程批量创建资源
    const createdResources = await batchCreateResourcesFromScan(scanResult);
    
    return {
      mode: 'new',
      createdResources,
      addedItems: createdResources.map((r, idx) => ({
        resourceId: r.id,
        guid: r.guid,
        title: r.title,
        sortOrder: this.calculateNextSortOrder()
      }))
    };
  }
}
```

### **2.3 排序交互组件**

```typescript
class ItemSorterUI {
  async sort(items: CollectionItem[]): Promise<SortedItemsResult> {
    console.log('\n┌─ 调整顺序 ───────────────────────────┐');
    console.log('│                                        │');
    
    // 可视化排序界面
    while (true) {
      this.renderSortableList(items);
      
      const action = await promptKeyAction([
        { key: 'up', action: () => this.moveItem(-1) },
        { key: 'down', action: () => this.moveItem(1) },
        { key: 'x', action: () => this.swapItems() },
        { key: 'enter', action: () => this.confirm() }
      ]);
      
      if (action === 'confirm') {
        break;
      }
    }
    
    return {
      success: true,
      sortedItems: items,
      orderChanged: this.hasOrderChanged()
    };
  }
  
  private renderSortableList(items: CollectionItem[]) {
    console.clear();
    
    console.log('\n┌─ 条目排序 ───────────────────────────┐');
    console.log('│                                      │');
    
    items.forEach((item, index) => {
      const isSelected = index === this.cursorIndex;
      const prefix = isSelected ? ui.color.primary : ui.color.disabled;
      const reset = ui.color.reset;
      
      console.log(`│ ${prefix}${String(index + 1).padStart(2)}. ${reset}${item.title}`);
    });
    
    console.log('│');
    console.log('│  ↑↓ 键移动选中标记                  │');
    console.log('│  UP/DOWN 键实际交换位置               │');
    console.log('│  X 键交换相邻两项                   │');
    console.log('│  ENTER 确认                         │');
    console.log('└──────────────────────────────────────┘\n');
  }
}
```

## 🚨 **三、异常分支处理**

### **3.1 错误码清单**

| 错误码 | 触发条件 | 用户提示 | 修复建议 |
|-------|---------|---------|---------|
| COLLECTION_NOT_FOUND | 合集不存在 | "指定的合集不存在" | 检查合集 ID |
| ITEM_ALREADY_EXISTS | 条目已存在 | "该资源已在合集中" | 跳过或移除旧条目 |
| DUPLICATE_GUID | GUID 重复 | "GUID 重复：{guid}" | 修改 GUID 值 |
| RSS_DERIVED_LOCKED | RSS 锁定字段被修改 | "无法移除 RSS 自动收录的条目" | 禁用 RSS 后重试 |
| SORT_ORDER_INVALID | 排序无效 | "排序索引越界" | 重新调整顺序 |
| BATCH_CREATE_FAILED | 创建资源失败 | "批量创建资源失败" | 查看详细错误 |

### **3.2 RSS 锁定保护**

```typescript
function checkRSSEpisodeProtection(
  item: CollectionItem,
  rssConfig: RSSConfig
): ProtectionResult {
  
  if (!item.isRSSDerived) {
    return { protected: false };
  }
  
  // 检查 GUID 是否在 RSS feed 中存在
  const latestFeed = fetchLatestRSSFeed(rssConfig.feedUrl);
  const feedEntry = latestFeed.entries.find(e => e.guid === item.guid);
  
  if (!feedEntry) {
    return {
      protected: true,
      reason: 'GUID 已从 RSS 中移除，无法手动删除',
      suggestion: '请先解除 RSS 绑定'
    };
  }
  
  // 检查是否需要更新条目信息
  const needsUpdate = shouldUpdateFromRSS(item, feedEntry);
  
  if (needsUpdate) {
    return {
      protected: true,
      reason: 'RSS 中已更新该条目信息',
      suggestion: '同步 RSS 后自动更新'
    };
  }
  
  return { protected: false };
}
```
