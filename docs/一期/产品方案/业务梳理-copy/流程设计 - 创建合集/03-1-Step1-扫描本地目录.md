### **2.1 扫描逻辑核心实现**

```typescript
// packages/console/src/pages/collection/creator/Step1/index.tsx
interface ScanConfig {
  rootDir: string;
  episodePatterns: string[];  // ['episode.json', 'feed.xml']
  ignorePatterns: string[];   // ['.git/*', '.DS_Store']
}

class Step1Controller {
  async scanCollection(config: ScanConfig): Promise<ScanResult> {
    // 递归扫描所有可能的 episode 文件
    const allFiles = await readdirRecursive(config.rootDir);
    
    // 识别 episode 文件并验证 GUID
    const episodes = allFiles
      .filter(f => config.episodePatterns.some(p => f.endsWith(p)))
      .map(filePath => parseEpisode(filePath))
      .filter((e): e is ValidEpisode => {
        if (!e.guid || !e.guid.trim()) {
          invalidEpisodes.push({ path: filePath, reason: 'MISSING_GUID' });
          return false;
        }
        return true;
      });
    
    // 检测集合类型
    const contentType = detectContentType(episodes);
    
    return { validEpisodes: episodes, invalidEpisodes, contentType };
  }
}
```

**业务规则**:
1. **GUID 必填验证**: 每个 episode 必须有唯一 GUID
2. **类型自动识别**: 根据文件结构和是否有 feed.xml 判断 Podcast/VideoSeries
3. **错误处理**: 解析失败的 episode 加入 invalidEpisodes 清单

```typescript
class ScanProgressRenderer {
  render(scanned: number, total: number, currentFile: string) {
    ui.clearLine();
    
    const percent = Math.round((scanned / total) * 100);
    
    console.log(`\r${ui.color.info}[${percent}%]${ui.color.reset} `);
    console.log(`${'▓'.repeat(Math.floor(percent / 10))}${
      '░'.repeat(10 - Math.floor(percent / 10))} ${scanned}/${total}`);
    
    if (currentFile) {
      console.log(ui.dim(`当前：${path.basename(currentFile)}`));
    }
  }
  
  showSummary(result: CollectionScanResult) {
    console.log('\n\n┌─ 扫描结果 ─────────────────────┐');
    console.log(`│ 有效条目：${result.validEpisodes.length}                    │`);
    console.log(`│ 无效条目：${result.invalidEpisodes.length}                   │`);
    console.log(`│ 总扫描文件：${result.statistics.totalFiles}                  │`);
    console.log(`│ 类型识别：${this.formatContentType(result.contentType)}      │`);
    console.log('└─────────────────────────────────┘\n');
    
    if (result.invalidEpisodes.length > 0) {
      console.log('⚠️  无效条目列表:');
      result.invalidEpisodes.slice(0, 5).forEach(entry => {
        console.log(`  ✗ ${entry.path}`);
        console.log(`    原因：${this.formatReason(entry.reason)}`);
      });
    }
    
    console.log('\n下一步：ENTER | 查看详细：D | 取消：ESC\n');
  }
  
  private formatContentType(type: CollectionType): string {
    const typeNames: Record<string, string> = {
      'podcast': '播客 Podcast',
      'video-series': '视频系列 Video Series',
      'audio-series': '音频系列 Audio Series',
      'static-collection': '静态合集 Static Collection'
    };
    return typeNames[type] || type;
  }
  
  private formatReason(reason: string): string {
    const reasons: Record<string, string> = {
      'MISSING_GUID': '缺少必填字段：guid',
      'PARSE_ERROR': 'JSON/YAML 解析失败',
      'DUPLICATE_GUID': 'GUID 重复',
      'UNSUPPORTED_FORMAT': '不支持的文件格式'
    };
    return reasons[reason] || reason;
  }
}
```

## 🚨 **三、异常分支处理**

| 异常场景 | 错误码 | 用户提示 | 处理方式 |
|---------|--------|---------|---------|
| 文件不是有效 JSON | MANIFEST_PARSE_ERROR | "无法解析 episode.json" | 加入 invalidEpisodes |
| 缺少必填字段 | MISSING_GUID | "缺少字段：guid" | 加入 invalidEpisodes |
| GUID 重复 | DUPLICATE_GUID | "发现重复 GUID" | 终止并提示修改 |
| 磁盘 I/O 错误 | DISK_IO_ERROR | "读取文件失败" | 尝试 3 次后标记为 invalid |
| 权限不足 | PERMISSION_DENIED | "无法访问：{path}" | 记录警告但继续扫描 |
