   import crypto from 'crypto';
   
   function calculateSHA1(filePath: string): Promise<string> {
     const hash = crypto.createHash('sha1');
     const stream = fs.createReadStream(filePath);
     
     return new Promise((resolve, reject) => {
       stream.on('data', data => hash.update(data));
       stream.on('end', () => resolve(hash.digest('hex')));
       stream.on('error', reject);
     });
   }

   async function validateCoverImage(file: File): Promise<ValidatedImage> {
     // 1. 格式验证
     if (!ALLOWED_FORMATS.includes(file.type)) {
       throw new ValidationError('COVER_FORMAT_INVALID', 
         '仅支持 JPG/PNG/WebP 格式');
     }
     
     // 2. 大小验证
     if (file.size > MAX_SIZE) {
       throw new ValidationError('COVER_SIZE_EXCEEDED',
         `封面图片不能大于${MAX_SIZE / 1024 / 1024}MB`);
     }
     
     // 3. 尺寸验证
     const dimensions = await getImageDimensions(file);
     if (dimensions.width < MIN_WIDTH || dimensions.height < MIN_HEIGHT) {
       throw new ValidationError('COVER_DIMENSIONS_INSUFFICIENT',
         `封面图片最小尺寸为${MIN_WIDTH}x${MIN_HEIGHT}`);
     }
     
     // 4. 质量检查 (检测过度压缩/模糊)
     const qualityScore = await analyzeImageQuality(file);
     if (qualityScore < MIN_QUALITY_THRESHOLD) {
       throw new ValidationError('COVER_QUALITY_LOW',
         '封面图片质量过低，请上传更清晰的图片');
     }
     
     return { file, sha256, dimensions, qualityScore };
   }

### **6.2 Checkpoint 保存时机**

| 时机 | 触发条件 | 保存内容 | 代码位置 |
|------|---------|---------|---------|
| **BEFORE_REMOTE_WRITE** | Step1 填写完成后 | resourceId, authId, type | `Step1/confirmHandler` |

### **6.3 Checkpoint 恢复逻辑**

```typescript
async function resumeCheckpoint(runId: string): Promise<ResumeState | null> {
  // 1. 读取本地 checkpoint 文件
  const checkpointPath = getCheckpointPath(runId);
  if (!fs.existsSync(checkpointPath)) {
    return null; // 无 checkpoint，从头开始
  }
  
  const data = readCheckpoint(checkpointPath);
  
  // 2. TTL 过期检查
  if (Date.now() > new Date(data.expiresAt).getTime()) {
    deleteCheckpoint(runId);
    warn('Checkpoint 已过期，将从头开始');
    return null;
  }
  
  // 3. accountId 严格匹配
  const currentAccount = await getCurrentAccount();
  if (data.accountId !== currentAccount.id) {
    error('账号不一致，丢弃旧 Checkpoint');
    debug(`期望：${data.accountId}, 当前：${currentAccount.id}`);
    return null;
  }
  
  // 4. 判断可以恢复的步骤
  let canResumeFrom: StepNumber = 1;
  if (data.completedData.step1_resourceShell) canResumeFrom = 2;
  if (data.completedData.step2_fileUpload) canResumeFrom = 3;
  if (data.completedData.step3_policy) canResumeFrom = 4;
  
  return {
    runId,
    canResumeFrom,
    completedData: data.completedData,
    metadata: {
      lastUpdatedAt: data.lastUpdatedAt,
      cliVersion: data.cliVersion
    }
  };
}
```

## 🚨 **七、异常分支与错误码**

### **7.1 核心异常分支清单**

| 异常场景 | 错误码 | 用户提示 | 修复建议 | 重试策略 |
|---------|--------|---------|---------|---------|
| 资源类型不存在 | RESOURCE_TYPE_NOT_FOUND | "所选资源类型不可用，请刷新重试" | 重新查询类型树 | 指数退避 (1s, 2s, 4s) |
| 授权标识重复 | AUTH_ID_EXISTS | "该授权标识已被使用" | 修改标题或手动修改 authId | 不允许自动重试 |
| 标题过长 | TITLE_TOO_LONG | "标题长度为{actual}/200，请缩短" | 精简标题 | 用户修正后手动重试 |
| 文件上传失败 | UPLOAD_TIMEOUT | "上传超时，网络可能不稳定" | 检查网络连接 | 断点续传 + 指数退避 |
| 封面格式错误 | COVER_FORMAT_INVALID | "不支持的图片格式" | 转换为 JPG/PNG/WebP | 不允许自动重试 |
| 封面尺寸不足 | COVER_DIMENSIONS_INSUFFICIENT | "封面图片太小，最小{minWidth}x{minHeight}" | 更换更大尺寸的封面 | 用户修正后手动重试 |
| 策略编译失败 | POLICY_COMPILE_ERROR | "策略编译失败：{reason}" | 检查策略参数是否合法 | 允许修改后重试 |
| API 速率限制 | RATE_LIMIT_EXCEEDED | "请求过于频繁，请稍后再试" | 等待后重试 | 指数退避 (60s, 120s, 300s) |
| 网络中断 | NETWORK_DISCONNECTED | "网络连接中断" | 检查网络 | 自动重连 |
| 磁盘空间不足 | DISK_SPACE_LOW | "磁盘空间不足，至少需要{required}GB" | 清理磁盘空间 | 不允许自动重试 |

### **7.2 错误处理实现**

```typescript
class ReleaseErrorHandler {
  async handle(error: unknown, step: StepNumber): Promise<HandleResult> {
    const errorCode = this.extractErrorCode(error);
    
    switch (errorCode) {
      case 'NETWORK_DISCONNECTED':
        return this.handleNetworkDisconnect(step);
      
      case 'AUTH_ID_EXISTS':
        return this.handleAuthIdConflict(step);
      
      case 'UPLOAD_TIMEOUT':
        return this.handleUploadTimeout(error, step);
      
      case 'RATE_LIMIT_EXCEEDED':
        return this.handleRateLimit(error, step);
      
      default:
        return this.handleUnknownError(error, step);
    }
  }
  
  private handleNetworkDisconnect(step: StepNumber): HandleResult {
    ui.showMessageBox({
      type: 'warning',
      message: '网络连接中断',
      details: '正在尝试重新连接...',
      actions: ['重试', '取消']
    });
    
    return {
      action: 'retry',
      strategy: 'exponential_backoff',
      initialDelay: 1000,
      maxDelay: 30000
    };
  }
  
  private handleAuthIdConflict(step: StepNumber): HandleResult {
    ui.showMessageBox({
      type: 'error',
      message: '授权标识已被使用',
      suggestions: [
        '修改标题以生成新的授权标识',
        '手动修改授权标识为其他值'
      ],
      actions: ['修改', '取消']
    });
    
    return {
      action: 'user_correction_required',
      preserveData: true // 保留已填写的其他数据
    };
  }
}
```

// StepIndicator 组件
class StepIndicator {
  render(currentStep: number, totalSteps: number) {
    const steps = [];
    
    for (let i = 1; i <= totalSteps; i++) {
      if (i === currentStep) {
        steps.push(ui.color.primary + `[${i}/${totalSteps}]` + ansi.reset);
      } else if (i < currentStep) {
        steps.push(ui.color.completed + `[✓]` + ansi.reset);
      } else {
        steps.push(ui.color.disabled + `[ ]` + ansi.reset);
      }
    }
    
    console.log(ui.box(['流程进度:', steps.join(' → ')]));
  }
}

class ProgressBar {
  private segments = 20;
  
  render(progress: number) {
    const filled = Math.round((progress / 100) * this.segments);
    const empty = this.segments - filled;
    
    const bar = ui.color.success + '▓'.repeat(filled) + 
                ui.color.info + '░'.repeat(empty) + ansi.reset;
    
    console.log(`${bar} ${Math.round(progress)}%`);
  }
