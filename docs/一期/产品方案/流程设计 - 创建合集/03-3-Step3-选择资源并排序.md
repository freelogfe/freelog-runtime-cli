## 🔍 **二、核心实现**

### **2.1 多选择器组件**

```typescript
class EpisodeMultiSelector {
  render(episodes: EpisodeEntry[], selectedGuids: Set<string>) {
    console.log('\n┌─ 选择条目 ─────────────────────────┐');
    
    episodes.forEach((episode, index) => {
      const isSelected = selectedGuids.has(episode.guid);
      const icon = isSelected ? '●' : '○';
      
      console.log(`│  ${icon} ${episode.title.padEnd(40)}`);
    });
    
    console.log('│   [全选] A | [反选] I | [确认] ENTER│');
    console.log(`│  当前选中 ${selectedGuids.size}/${episodes.length} 个条目    │`);
    console.log('└────────────────────────────────────┘\n');
  }
}
```

### **2.2 排序界面**

```typescript
class EpisodeSorter {
  render(sortedEpisodes: EpisodeEntry[]) {
    console.log('\n┌─ 调整顺序 ─────────────────────────┐');
    
    sortedEpisodes.forEach((episode, index) => {
      console.log(`│ ${String(index + 1).padStart(2)}. ${episode.title}`);
    });
    
    console.log('│   ← ↑ ↓ → 键调整位置 | ENTER 确认   │');
    console.log('└────────────────────────────────────┘\n');
  }
}
```

### **2.3 GUID 去重检测**

```typescript
async function detectDuplicateGuids(episodes: EpisodeEntry[]): Promise<DuplicateInfo[]> {
  const guidMap = new Map<string, string>();
  const duplicates: DuplicateInfo[] = [];
  
  for (const episode of episodes) {
    if (guidMap.has(episode.guid)) {
      duplicates.push({
        guid: episode.guid,
        paths: [guidMap.get(episode.guid)!, episode.path]
      });
    } else {
      guidMap.set(episode.guid, episode.path);
    }
  }
  
  return duplicates;
}
```
