## 🔍 **二、Step1: 扫描本地目录**

### **2.1 扫描规则定义**

```typescript
interface CollectionScanRules {
  // 入口目录
  rootDir: string;
  
  // 匹配规则
  episodePatterns: string[];  // ['episode.json', 'feed.xml']
  feedPatterns: string[];     // ['feed.xml', 'rss.xml']
  ignorePatterns: string[];   // ['.git/*', '.DS_Store', 'node_modules/**']
  
  // 过滤选项
  skipHiddenFiles: true;
  skipInvalidGuids: false;    // 是否跳过无 GUID 的条目
}

class CollectionScanner {
  async scan(config: ScanConfig): Promise<CollectionScanResult> {
    const episodes: EpisodeEntry[] = [];
    const invalidEpisodes: InvalidEntry[] = [];
    
    // 1. 递归扫描所有可能的 episode 文件
    const allFiles = await this.readdirRecursive(config.rootDir);
    
    // 2. 识别 episode 文件和 feed 文件
    const episodeFiles = allFiles.filter(f => 
      config.episodePatterns.some(p => f.endsWith(p))
    );
    
    const feedFile = allFiles.find(f => 
      config.feedPatterns.some(p => f.endsWith(p))
    );
    
    // 3. 解析每个 episode
    for (const filePath of episodeFiles) {
      try {
        const episode = await this.parseEpisode(filePath);
        
        // GUID 是唯一必需字段
        if (!episode.guid || episode.guid.length === 0) {
          invalidEpisodes.push({
            path: filePath,
            reason: 'MISSING_GUID',
            errorMessage: 'episode 缺少必填字段：guid'
          });
          continue;
        }
        
        episodes.push(episode);
      } catch (err) {
        invalidEpisodes.push({
          path: filePath,
          reason: 'PARSE_ERROR',
          errorMessage: err.message
        });
      }
    }
    
    // 4. 检测是否为播客类型
    const contentType = this.detectContentType(episodes, feedFile);
    
    return {
      validEpisodes: episodes,
      invalidEpisodes,
      hasFeedFile: !!feedFile,
      contentType,  // 'podcast' / 'video-series' / 'audio-series' / 'unknown'
      statistics: this.computeStatistics(episodes, invalidEpisodes)
    };
  }
  
  private detectContentType(
    episodes: EpisodeEntry[],
    feedFile: string | undefined
  ): CollectionType {
    if (!feedFile && episodes.every(e => !e.mediaType)) {
      return 'static-collection';
    }
    
    if (feedFile && feedFile.includes('rss')) {
      return 'podcast';
    }
    
    return 'dynamic-collection';
  }
}
```

### **2.2 Episode 解析逻辑**

```typescript
async function parseEpisode(filePath: string): Promise<EpisodeEntry> {
  const content = fs.readFileSync(filePath, 'utf-8');
  const ext = path.extname(filePath);
  
  let episodeData: EpisodeSchema;
  
  switch (ext) {
    case '.json':
      episodeData = JSON.parse(content);
      break;
      
    case '.xml':
    case '.rss':
      episodeData = parseRssItem(content);
      break;
      
    default:
      throw new ValidationError('UNSUPPORTED_FORMAT', '不支持的 episode 格式');
  }
  
  // 验证必填字段
  const validated = EpisodeSchema.parse(episodeData);
  
  return {
    path: filePath,
    directory: path.dirname(filePath),
    guid: validated.guid!,
    title: validated.title,
    description: validated.description,
    mediaType: validated.mediaType || 'audio',
    pubDate: validated.pubDate ? new Date(validated.pubDate) : null,
    duration: validated.duration || null,
    author: validated.author || null,
    isValid: true
  };
}

// RSS 解析器
function parseRssItem(rssContent: string): EpisodeSchema {
  const parser = new DOMParser();
  const xmlDoc = parser.parseFromString(rssContent, "text/xml");
  
  const item = xmlDoc.querySelector('item');
  if (!item) {
    throw new Error('RSS 中未找到 episode 条目');
  }
  
  return {
    guid: item.querySelector('guid')?.textContent || '',
    title: item.querySelector('title')?.textContent || '',
    description: item.querySelector('description')?.textContent || '',
    pubDate: item.querySelector('pubDate')?.textContent || '',
    enclosure: {
      url: item.querySelector('enclosure@url')?.textContent || '',
      type: item.querySelector('enclosure@type')?.textContent || '',
      length: item.querySelector('enclosure@length')?.textContent || ''
    }
  };
}
```

```typescript
async function autoFillCollectionInfo(rootDir: string): Promise<Partial<CollectionSchema>> {
  // 1. 尝试读取 package.json
  const packageJsonPath = path.join(rootDir, 'package.json');
  if (fs.existsSync(packageJsonPath)) {
    const pkg = JSON.parse(fs.readFileSync(packageJsonPath, 'utf-8'));
    
    return {
      title: pkg.name || path.basename(rootDir),
      description: pkg.description || '',
      author: pkg.author || null,
      website: pkg.homepage || null
    };
  }
  
  // 2. 如果没有 package.json，从根目录第一个 episode 读取
  const firstEpisode = await findFirstEpisode(rootDir);
  if (firstEpisode) {
    return {
      title: firstEpisode.title,
      description: firstEpisode.description.substring(0, 100),
      author: firstEpisode.author
    };
  }
  
  // 3. 完全空对象，让用户手动填写
  return {};
}
```

### **5.2 RSS 绑定完整流程**

```typescript
class RSSBindingFlow {
  async bind(feedUrl: string): Promise<RSSBindingResult> {
    // 1. Feed URL 验证
    const urlValidation = await this.validateFeedUrl(feedUrl);
    if (!urlValidation.valid) {
      throw new ValidationError('INVALID_FEED_URL', urlValidation.reason);
    }
    
    // 2. 检测是否需要验证码
    const captchaRequired = await this.checkCaptchaRequired(feedUrl);
    
    let captchaResponse: string | undefined;
    if (captchaRequired) {
      console.log('⚠️  该 feed 需要验证码验证');
      captchaResponse = await promptCaptchaInput();
      
      const captchaValid = await this.verifyCaptcha(feedUrl, captchaResponse);
      if (!captchaValid) {
        throw new ValidationError('CAPTCHA_INVALID', '验证码错误');
      }
    }
    
    // 3. 获取 feed 元数据
    const feedMetadata = await this.fetchFeedMetadata(feedUrl);
    
    // 4. 检查已有 GUID 的差异
    const existingGuids = await this.getExistingCollectionGuids();
    const newGuidsInFeed = await this.extractNewGuidsFromFeed(feedUrl);
    
    const diff = this.computeGuidDiff(existingGuids, newGuidsInFeed);
    if (diff.mismatchCount > 0) {
      console.log('⚠️  检测到 GUID 差异:');
      console.log(`   新增 GUID: ${diff.newGuids.length}`);
      console.log(`   修改 GUID: ${diff.modifiedGuids.length}`);
      console.log('   → 建议：使用 GUID 作为唯一标识避免重复');
    }
    
    // 5. 数量限制检查
    if (newGuidsInFeed.length > MAX_EPISODES_PER_COLLECTION) {
      warn(`Feed 包含 ${newGuidsInFeed.length} 个 episode,超过单个合集最大限制 ${MAX_EPISODES_PER_COLLECTION}`);
      warn('建议：分多个合集或只同步最近 N 天的内容');
    }
    
    // 6. 锁定字段定义
    const lockedFields: LockableField[] = [
      'guid',
      'pubDate',
      'enclosure.url'
    ];
    
    return {
      feedUrl,
      feedTitle: feedMetadata.title,
      feedAuthor: feedMetadata.author,
      episodeCount: newGuidsInFeed.length,
      captchaVerified: !!captchaResponse,
      lockedFields,
      warnings: diff.mismatchCount > 0 ? ['GUID_MISMATCH_DETECTED'] : []
    };
  }
}
```

```typescript
async function submitCollection(
  collection: CollectionSubmission
): Promise<SubmitResult> {
  // POST /api/collection/create
  const response = await api.collection.create({
    // 基本信息
    title: collection.title,
    description: collection.description,
    coverImage: collection.coverImage,
    
    // 资源关系
    episodes: collection.episodes.map(e => ({
      guid: e.guid,
      resourceId: e.resourceId,
      sortOrder: e.sortOrder
    })),
    
    // 收录规则
    rule: {
      type: collection.rule.type,
      ...(collection.rule.type === 'rss-dynamic' ? {
        feedUrl: collection.rule.feedUrl,
        syncInterval: collection.rule.syncInterval,
        maxEpisodesPerSync: collection.rule.maxEpisodesPerSync,
        daysBackFilter: collection.rule.daysBackFilter,
        lockedFields: collection.rule.lockedFields
      } : {})
    },
    
    // 策略
    strategy: {
      templateId: collection.strategy.templateId,
      parameters: collection.strategy.parameters,
      compiledPolicy: collection.strategy.compiledPolicy
    },
    
    // Listing
    listing: {
      tags: collection.tags,
      website: collection.website,
      contactEmail: collection.contactEmail
    },
    
    // 发布选项
    autoPublish: collection.autoPublish,
    notifySubscribers: true
  });
  
  return {
    success: true,
    collectionId: response.collectionId,
    versionNumber: response.versionNumber,
    status: response.status, // 'published' | 'pending_review'
    reviewEstimate: response.reviewEstimate // '24-48 hours'
```

### **7.2 恢复策略**

```typescript
async function resumeCollectionCheckpoint(
  runId: string
): Promise<ResumeState | null> {
  
  const checkpoint = readCheckpoint(runId);
  
  // TTL 检查
  if (Date.now() > new Date(checkpoint.expiresAt).getTime()) {
    return null;
  }
  
  // accountId 严格匹配
  const currentAccount = await getCurrentAccount();
  if (checkpoint.accountId !== currentAccount.id) {
    return null;
  }
  
  // 计算可以恢复的步骤
  let canResumeFrom = 1;
  if (checkpoint.step1_scan) canResumeFrom = 2;
  if (checkpoint.step2_metadata) canResumeFrom = 3;
  if (checkpoint.step3_selection) canResumeFrom = 4;
  if (checkpoint.step4_rss) canResumeFrom = 5;
  
  return {
    runId,
    canResumeFrom,
    completedSteps: {
      scan: checkpoint.step1_scan,
      metadata: checkpoint.step2_metadata,
      selection: checkpoint.step3_selection,
      rss: checkpoint.step4_rss,
      listing: checkpoint.step5_listing
    }
  };
}
```

## 🚨 **八、异常分支与错误码**

| 错误码 | 触发条件 | 用户提示 | 修复建议 |
|-------|---------|---------|---------|
| COLLECTION_SCAN_FAILED | 扫描过程中断 | "目录扫描失败" | 检查磁盘权限 |
| MISSING_GUID | episode 缺少 GUID | "条目缺少必填字段：guid" | 补充 guid 字段 |
| DUPLICATE_GUID | GUID 重复 | "发现重复 GUID: {guid}" | 修改 GUID 值 |
| FEED_URL_INVALID | RSS URL 无效 | "无效的 Feed URL" | 检查 URL 格式 |
| CAPTCHA_REQUIRED | feed 需要验证码 | "Feed 需要验证码验证" | 输入验证码 |
| RSS_SYNC_LIMIT_EXCEEDED | 同步数量超限 | "一次性同步过多，限制 50 集" | 增加天数范围 |
| COLLECTION_LIMIT_REACHED | 单合集条目超限 | "条目数超过 2000 限制" | 拆分多个合集 |
