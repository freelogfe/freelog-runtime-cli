## 🔍 **二、详细步骤实现**

### **2.1 自动填充逻辑**

```typescript
async function autoFillCollectionInfo(rootDir: string): Promise<Partial<CollectionSchema>> {
  // 1. 尝试读取 package.json
  const packageJsonPath = path.join(rootDir, 'package.json');
  if (fs.existsSync(packageJsonPath)) {
    const pkg = JSON.parse(fs.readFileSync(packageJsonPath, 'utf-8'));
    
    return {
      title: pkg.name || path.basename(rootDir),
      description: pkg.description || '',
      author: pkg.author || null,
      website: pkg.homepage || null
    };
  }
  
  // 2. 如果没有 package.json，从第一个 episode 读取
  const firstEpisode = await findFirstEpisode(rootDir);
  if (firstEpisode) {
    return {
      title: firstEpisode.title,
      description: firstEpisode.description.substring(0, 100),
      author: firstEpisode.author
    };
  }
  
  // 3. 完全空对象，让用户手动填写
  return {};
}
```

### **2.2 TTY 表单组件**

```typescript
class CollectionMetaForm {
  render(collection: Partial<CollectionSchema>) {
    console.log('\n┌─ 合集信息 ─────────────────────────┐');
    console.log('│                                     │');
    
    // 标题输入
    this.inputField('合集标题 *', {
      value: collection.title || '',
      placeholder: '默认基于 package.json 自动生成',
      validate: (val: string) => val.length > 0 && val.length <= 200 ? null : '标题不能为空且不超过 200 字符'
    });
    
    // 描述输入
    this.textAreaField('描述 *', {
      value: collection.description || '',
      maxLength: 1000,
      minLength: 50,
      validate: (val: string) => {
        if (val.length < 50) return `描述至少需要${50}个字符`;
        if (val.length > 1000) return `描述不能超过${1000}个字符`;
        return null;
      }
    });
    
    // 封面上传
    if (collection.coverImage) {
      console.log('│ 封面图片                          │');
      console.log(`│ ✓ ${collection.coverImage.fileName} (${formatSize(collection.coverImage.size)})`);
      console.log(`│   尺寸：${collection.coverImage.width}x${collection.coverImage.height}`);
    } else {
      this.imageUploadField('封面图片', {
        placeholder: '点击上传封面 (推荐尺寸：1200x628px)',
        acceptedFormats: ['image/jpeg', 'image/png', 'image/webp'],
        maxSize: 5 * 1024 * 1024
      });
    }
    
    console.log('└─────────────────────────────────────┘\n');
  }
}
```

### **2.3 封面图片验证**

```typescript
async function validateCoverImage(file: File): Promise<ValidatedImage> {
  // 1. 格式验证
  if (!['image/jpeg', 'image/png', 'image/webp'].includes(file.type)) {
    throw new ValidationError('COVER_FORMAT_INVALID', 
      '仅支持 JPG/PNG/WebP 格式');
  }
  
  // 2. 大小验证
  if (file.size > 5 * 1024 * 1024) {
    throw new ValidationError('COVER_SIZE_EXCEEDED',
      '封面图片不能大于 5MB');
  }
  
  // 3. 尺寸验证
  const dimensions = await getImageDimensions(file);
  if (dimensions.width < 800 || dimensions.height < 600) {
    throw new ValidationError('COVER_DIMENSIONS_INSUFFICIENT',
      '封面图片最小尺寸为 800×600');
  }
  
  const sha256 = await calculateSHA256(file.path);
  
  return { file, sha256, dimensions, qualityScore: 0.9 };
}
```

## 🚨 **三、异常分支处理**

| 错误码 | 触发条件 | 用户提示 | 修复建议 |
|-------|---------|---------|---------|
| MISSING_PACKAGE_JSON | package.json 不存在 | "未找到 package.json" | 补充元数据后重试 |
| COVER_FORMAT_INVALID | 封面格式不支持 | "不支持的图片格式" | 转换为 JPG/PNG/WebP |
| COVER_SIZE_EXCEEDED | 封面超过 5MB | "封面图片太大，不超过 5MB" | 压缩图片尺寸 |
| DESCRIPTION_TOO_SHORT | 描述太短 | "描述至少需要 50 个字符" | 补充描述内容 |
