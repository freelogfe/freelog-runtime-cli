### **2.1 状态查询 API**

```typescript
interface ResourceVersionInfo {
  resourceId: string;
  currentVersion: {
    versionNumber: string;
    fileId: string;
    fileSize: number;
    sha1: string;
    publishedAt: string;
    status: 'published' | 'pending_review' | 'draft';
  };
  metadata: {
    title: string;
    description: string;
    tags: string[];
    author: string;
  };
  assets: {
    coverImage?: ImageRef;
    policyId: string;
  };
  status: {
    isFrozen: boolean;
    canUpdate: boolean;
    frozenReason?: string;
  };
}

async function fetchResourceVersionInfo(resourceId: string): Promise<ResourceVersionInfo> {
  const response = await api.resource.get({ resourceId });
  
  // 检查是否可以更新
  if (response.status.isFrozen && !response.status.canUpdate) {
    throw new ValidationError('RESOURCE_FROZEN', `资源已被冻结，原因：${response.status.frozenReason}`);
  }
  
  return response;
}
```

### **3.2 文件对比逻辑**

```typescript
async function compareFiles(
  oldFile: FileInfo,
  newFile: File
): Promise<FileComparison> {
  
  const comparison: FileComparison = {
    identical: false,
    sizeDiff: newFile.size - oldFile.size,
    typeDiff: false,
    recommendedAction: 'upload-new'
  };
  
  // 计算新文件 SHA1
  const newSHA1 = await calculateSHA1(newFile.path);
  
  // 比较 SHA1
  if (newSHA1 === oldFile.sha1) {
    comparison.identical = true;
    comparison.recommendedAction = 'skip-upload';
  }
  
  // 判断文件大小变化率
  const sizeRatio = Math.abs(comparison.sizeDiff) / oldFile.size;
  if (sizeRatio > 0.5) {
    comparison.majorChange = true; // 大于 50% 变化
  }
  
  return comparison;
}

// TTY 显示文件对比结果
function showFileComparison(comparison: FileComparison) {
  if (comparison.identical) {
    console.log('ℹ️  新旧文件完全相同，无需上传');
    console.log('   将直接复用旧版本的 fileID');
    return;
  }
  
  if (comparison.majorChange) {
    console.log('⚠️  文件发生较大变化 (>50%)');
    console.log(`   大小变化：${formatBytes(comparison.sizeDiff)}`);
  } else {
    console.log('📝  文件有较小改动');
  }
}
```

### **5.1 语义化版本自动递增**

```typescript
function calculateNextVersion(
  currentVersion: string,
  changeType: 'patch' | 'minor' | 'major'
): string {
  const [major, minor, patch] = currentVersion
    .replace(/^v/, '')
    .split('.')
    .map(Number);
  
  switch (changeType) {
    case 'patch':
      return `${major}.${minor}.${patch + 1}`;
    case 'minor':
      return `${major}.${minor + 1}.0`;
    case 'major':
      return `${major + 1}.0.0`;
    default:
      return currentVersion;
  }
}

// ChangeType 映射表
const CHANGE_TYPE_MAPPING = {
  'fix': 'patch',
  'bugfix': 'patch',
  'patch': 'patch',
  'feat': 'minor',
  'feature': 'minor',
  'breaking': 'major',
  'breaking-change': 'major'
};
```

```typescript
interface ChangelogValidationRules {
  minLength: 20;    // 最少 20 字符
  maxLength: 2000;  // 最多 2000 字符
  requiredFormat: true;  // 必须使用列表格式
  suggestions: string[];
}

function validateChangelog(text: string): ValidationResult {
  const charCount = text.length;
  
  if (charCount < 20) {
    return error(`变更日志至少需要${20}个字符 (当前${charCount})`);
  }
  
  if (charCount > 2000) {
    return error(`变更日志不能超过${2000}个字符`);
  }
  
  // 检测是否使用列表格式 (- 或 * 开头)
  const lines = text.split('\n').filter(l => l.trim());
  const hasListFormat = lines.some(l => /^\s*[-*]\s/.test(l));
  
  if (!hasListFormat) {
    warn('建议：使用列表格式说明变更 (每行以 - 或 * 开头)');
  }
  
  return success();
}
```

```
│   → 继续使用 v1.0.0 的商业使用策略     │
│                                        │
│ 下一步：ENTER | 预览：P                │
└────────────────────────────────────────┘
```

**业务规则**:
1. **版本号自动生成**:
   - 根据变更类型 (patch/minor/major) 自动计算
   - 提供手动编辑入口
   - 符合语义化版本规范 (SemVer)

2. **变更日志要求**:
   - 最小长度：20 字符
   - 最大长度：2000 字符
   - 建议使用列表格式 (- 或 * 开头)
   - 自动检测并提供格式化提示

3. **继承策略选项**:
   - 默认勾选“继承旧版策略”(最常用)
   - 可以选择不继承，改为重新配置
   - **注意**: Version Update 不需要完整策略流程，只需简单继承决策

```typescript
async function submitVersionUpdate(
  updateRequest: VersionUpdateSubmission
): Promise<SubmitResult> {
  
  // POST /api/resource/version/create
  const response = await api.resource.createVersion({
    resourceId: updateRequest.resourceId,
    
    // 继承逻辑
    inheritFromPrevious: {
      metadata: updateRequest.inheritedMetadata,  // {title, tags, author}
      coverImage: updateRequest.inheritCoverImage,
      policy: updateRequest.inheritPolicy
    },
    
    // 新版本数据
    file: updateRequest.file ? {
      uploadId: updateRequest.uploadId,
      sha1: updateRequest.sha1
    } : undefined,
    
    listing: {
      versionNumber: updateRequest.versionNumber,
      changelog: updateRequest.changelog,
      ...(updateRequest.coverImage && { coverImage: updateRequest.coverImage }),
      tags: updateRequest.tags,
      title: updateRequest.title  // 如果改动了标题
    },
    
    strategy: updateRequest.strategy ? {
      templateId: updateRequest.strategy.templateId,
      parameters: updateRequest.strategy.parameters,
      compiledPolicy: updateRequest.strategy.compiledPolicy
    } : undefined,
    
    autoPublish: updateRequest.autoPublish
  });
  
  return {
    success: true,
    versionId: response.versionId,
    versionNumber: response.versionNumber,
    previousVersion: updateRequest.previousVersion,
    status: response.status,
    reviewEstimate: '24-48 hours'
  };
}
```

## 🚨 **六、异常分支与错误码**

| 错误码 | 触发条件 | 用户提示 | 修复建议 |
|-------|---------|---------|---------|
| RESOURCE_FROZEN | 资源被冻结 | "资源已被冻结，无法更新" | 联系平台解锁 |
| VERSION_NOT_FOUND | 版本不存在 | "指定的版本不存在" | 检查 resourceId |
| FILE_IDENTICAL | 新旧文件相同 | "新旧文件完全相同，无需上传" | 跳过上传步骤 |
| UPLOAD_TIMEOUT | 上传超时 | "上传超时，请检查网络" | 重试或断点续传 |
| POLICY_VERSION_MISMATCH | 策略版本冲突 | "策略模板版本不兼容" | 使用更新的模板 |
| CHANGELOG_TOO_SHORT | 变更日志过短 | "变更日志至少需要 20 个字符" | 补充详细信息 |
| AUTO_PUBLISH_REJECTED | 自动上架被拒 | "自动上架审核失败" | 改为仅提交审核 |
