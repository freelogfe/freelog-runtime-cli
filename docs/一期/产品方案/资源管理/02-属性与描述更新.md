### **2.1 TTY 表单组件**

```typescript
class ResourceMetadataForm {
  render(currentMetadata: ResourceMetadata) {
    console.log('\n┌─ 属性与描述更新 ────────────────────┐');
    console.log('│                                      │');
    
    // 标题输入
    this.inputField('标题 *', {
      value: currentMetadata.title,
      maxLength: 200,
      counter: true,
      validate: (val: string) => {
        if (val.length === 0) return '标题不能为空';
        if (val.length > 200) return '标题不能超过 200 字符';
        return null;
      }
    });
    
    // 描述文本域
    this.textAreaField('描述 *', {
      value: currentMetadata.description || '',
      minLength: 50,
      maxLength: 1000,
      counter: true,
      validate: (val: string) => {
        if (val.length < 50) return `描述至少需要${50}个字符`;
        if (val.length > 1000) return `描述不能超过${1000}个字符`;
        return null;
      }
    });
    
    // 封面图片选项
    this.imageChoiceField('封面图片', {
      currentImage: currentMetadata.coverImage,
      options: [
        { value: 'keep', label: '保持现有封面', default: true },
        { value: 'upload-new', label: '重新上传封面' }
      ]
    });
    
    // 标签输入
    this.tagInput('标签', {
      tags: currentMetadata.tags,
      maxTags: 10,
      autoNormalize: true,
      validate: (tags: string[]) => {
        if (tags.length === 0) return '至少需要一个标签';
        if (tags.length > 10) return '最多 10 个标签';
        return null;
      }
    });
    
    console.log('│\n│ [提交更新] ENTER | [取消] ESC         │');
    console.log('└──────────────────────────────────────┘\n');
  }
  
  private inputField(label: string, options: InputOptions) {
    const reset = ui.color.reset;
    const gray = ui.color.disabled;
    
    console.log(`│ ${gray}✎${reset} ${label.padEnd(18)} ${gray}(${options.maxLength}字符)${reset}`);
    console.log(`│ ┌${'─'.repeat(37)}┐`);
    console.log(`│ │ ${(options.value || '').padEnd(37)} │`);
    console.log(`│ └${'─'.repeat(37)}┘`);
  }
  
  private textAreaField(label: string, options: TextAreaOptions) {
    console.log(`│ ${ui.color.disabled}✎${ui.color.reset} ${label.padEnd(18)} (${options.minLength}-${options.maxLength}字符)`);
    console.log(`│ ┌${'─'.repeat(37)}┐`);
    
    const lines = options.value.split('\n');
    lines.forEach((line, idx) => {
      const padding = options.maxLength - line.length;
      if (idx < 3) {
        console.log(`│ │ ${line.padEnd(37)} │`);
      } else if (idx === 3 && lines.length > 3) {
        console.log(`│ │ ... ${'(更多行)'.padEnd(28)} │`);
      }
    });
    
    console.log(`│ └${'─'.repeat(37)}┘`);
  }
}
```

### **2.2 标签标准化逻辑**

```typescript
class TagNormalizer {
  normalize(rawTags: string): NormalizedTag[] {
    return rawTags
      .split(',')
      .map(tag => tag.trim())
      .filter(tag => tag.length > 0 && tag.length <= 50)
      .map(tag => tag.toLowerCase())
      .filter((tag, index, arr) => arr.indexOf(tag) === index)
      .map(tag => ({
        id: this.generateTagId(tag),
        name: tag,
        normalized: this.normalizeForSearch(tag)
      }));
  }
  
  private generateTagId(tag: string): string {
    return 'tag-' + Buffer.from(tag).toString('hex');
  }
  
  private normalizeForSearch(tag: string): string {
    // 移除特殊字符，转小写，去除空格
    return tag.replace(/[^a-z0-9]/g, '').toLowerCase();
  }
  
  detectDuplicates(tags: NormalizedTag[]): DuplicateInfo[] {
    const duplicates: DuplicateInfo[] = [];
    
    for (let i = 0; i < tags.length; i++) {
      for (let j = i + 1; j < tags.length; j++) {
        if (this.areSimilar(tags[i].normalized, tags[j].normalized)) {
          duplicates.push({
            original: [tags[i].name, tags[j].name],
            reason: 'SIMILAR_TAGS',
            suggestion: this.mergeSuggestion(tags[i], tags[j])
          });
        }
      }
    }
    
    return duplicates;
  }
  
  private areSimilar(normalized1: string, normalized2: string): boolean {
    // 检测相似标签 (如 "theme" vs "themes")
    const plurals = ['s', 'es'];
    for (const plural of plurals) {
      if (normalized1.endsWith(plural) && 
          normalized2 === normalized1.slice(0, -plural.length)) {
        return true;
      }
    }
    return false;
  }
}
```

## 🚨 **三、异常分支处理**

### **3.1 错误码清单**

| 错误码 | 触发条件 | 用户提示 | 修复建议 |
|-------|---------|---------|---------|
| METADATA_UPDATE_FAILED | API 调用失败 | "属性更新失败" | 检查网络连接后重试 |
| TITLE_TOO_LONG | 标题超过 200 字符 | "标题长度为{actual}/200，请缩短" | 精简标题 |
| DESCRIPTION_TOO_SHORT | 描述少于 50 字符 | "描述至少需要 50 个字符" | 补充详细说明 |
| TAG_TOO_MANY | 超过 10 个标签 | "最多只能有 10 个标签" | 删除不重要的标签 |
| DUPLICATE_TAGS | 发现重复或相似标签 | "检测到重复标签：{tags}" | 合并相似标签 |
| COVER_UPLOAD_FAILED | 封面上传失败 | "封面图片上传失败" | 检查文件格式/大小后重新上传 |

### **4.1 更新提交逻辑**

```typescript
async function submitMetadataUpdate(
  resourceId: string,
  updateRequest: MetadataUpdateSubmission
): Promise<SubmitResult> {
  
  // POST /api/resource/metadata/update
  const response = await api.resource.updateMetadata({
    resourceId,
    
    updates: {
      title: updateRequest.title !== undefined ? {
        value: updateRequest.title,
        validated: true
      } : undefined,
      
      description: updateRequest.description !== undefined ? {
        value: updateRequest.description,
        validated: true
      } : undefined,
      
      tags: updateRequest.tags?.map(tag => ({
        tagName: tag.name,
        normalized: tag.normalized
      })) ?? undefined,
      
      coverImage: updateRequest.newCoverImage || null
    }
  });
  
  return {
    success: true,
    updatedFields: Object.keys(response.updates).filter(k => 
      response.updates[k].hasChanged
    ),
    newVersion: response.versionNumber,
    syncStatus: 'synced'
  };
}
```

### **4.2 变更摘要生成**

```typescript
function generateChangeSummary(
  oldMetadata: ResourceMetadata,
  newMetadata: Partial<ResourceMetadata>
): ChangeSummary {
  
  const summary: ChangeSummary = {
    fieldsChanged: [],
    fieldDetails: {}
  };
  
  // 比较标题
  if (oldMetadata.title !== newMetadata.title) {
    summary.fieldsChanged.push('title');
    summary.fieldDetails.title = {
      oldValue: oldMetadata.title,
      newValue: newMetadata.title,
      changeType: 'modified'
    };
  }
  
  // 比较描述
  if (oldMetadata.description !== newMetadata.description) {
    summary.fieldsChanged.push('description');
    summary.fieldDetails.description = {
      oldValue: oldMetadata.description.substring(0, 50) + '...',
      newValue: newMetadata.description.substring(0, 50) + '...',
      changeType: 'modified'
    };
  }
  
  // 比较封面
  if (newMetadata.coverImage && oldMetadata.coverImage?.fileId !== newMetadata.coverImage.fileId) {
    summary.fieldsChanged.push('coverImage');
    summary.fieldDetails.coverImage = {
      oldValue: oldMetadata.coverImage.fileName,
      newValue: newMetadata.coverImage.fileName,
      changeType: 'updated'
    };
  }
  
  // 比较标签
  const oldTags = new Set(oldMetadata.tags);
  const newTags = new Set(newMetadata.tags.map(t => t.name));
  
  const addedTags = Array.from(newTags).filter(t => !oldTags.has(t));
  const removedTags = Array.from(oldTags).filter(t => !newTags.has(t));
  
  if (addedTags.length > 0 || removedTags.length > 0) {
    summary.fieldsChanged.push('tags');
    summary.fieldDetails.tags = {
      added: addedTags,
      removed: removedTags
    };
  }
  
  return summary;
}
```
