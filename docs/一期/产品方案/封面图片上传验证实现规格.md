### **1.1 流程图**

```text
进入封面配置阶段
  ↓
[Step 1] 选择封面来源
  ├─ 选项 A: 本地文件上传
  │   ├─ 用户选择本地图片文件
  │   └─ 读取文件进行校验
  │
  └─ 选项 B: URL 引用 (仅已上传过的图片)
      ├─ 输入或粘贴 URL
      └─ 直接保存到平台 (不重新上传)
  ↓
[Step 2] 格式验证 (本地文件模式)
  ├─ 检查文件扩展名 (.jpg/.jpeg/.png/.gif)
  ├─ 检查 MIME 类型 (文件头签名验证)
  └─ 失败则拒绝并显示具体原因
  ↓
[Step 3] 文件大小验证
  ├─ 检查文件字节大小 ≤ 5MB
  ├─ 显示当前大小和上限
  └─ 超限则给出压缩建议或拒绝
  ↓
[Step 4] 图像维度检测
  ├─ 解析图片元数据获取宽高
  ├─ 计算长宽比 (width/height)
  ├─ 对比推荐比例 (横图 16:9, 竖图 9:16)
  └─ 显示当前比例和建议，不阻止上传但高亮提示
  ↓
[Step 5] 图像质量检查 (可选增强)
  ├─ 检测是否过于模糊 (边缘检测方差)
  ├─ 检测是否过暗/过亮 (直方图统计)
  └─ 给出优化建议但不阻断
  ↓
[Step 6] 压缩优化 (可选)
  ├─ 如果原始 > 2MB 且用户允许
  ├─ 调用 sharp 进行有损压缩
  ├─ 目标质量：85%
  └─ 保持原尺寸，仅减小体积
  ↓
[Step 7] TTY 预览确认
  ├─ ASCII 艺术展示缩略图框架
  ├─ 显示图片关键信息
  │   ├─ 文件名、尺寸、体积、格式
  │   ├─ 长宽比、推荐比例
  │   └─ 是否满足推荐标准
  ├─ 用户确认或更换
  └─ 确认则继续上传

[Step 8] 调用 Storage.uploadImage
  ├─ 传递 Buffer(本地文件) 或 URL(引用)

  ├─ 保存至 checkpoint
  └─ 失败则提示重试或更换
  ↓
完成
  └─ 封面 URL 写入资源 listing 数据
```

#### **Step 2: 格式验证详细规则**

```typescript
interface ImageFormatValidator {
  // 支持的格式列表
  supportedFormats = ['jpeg', 'png', 'gif'] as const;
  
  // 对应的 MIME 类型
  mimeTypes = {
    jpeg: 'image/jpeg',
    png: 'image/png',
    gif: 'image/gif'
  } as const;
  
  // 对应文件扩展名
  extensions = {
    jpeg: ['.jpg', '.jpeg'],
    png: ['.png'],
    gif: ['.gif']
  } as const;
  
  // 文件格式头验证 (Magic Number)
  validateByMagicNumber: (buffer: Buffer): ImageFormat => {
    // JPEG: FF D8 FF
    if (buffer.length >= 3 && 
        buffer.readUInt8(0) === 0xFF && 
        buffer.readUInt8(1) === 0xD8 && 
        buffer.readUInt8(2) === 0xFF) {
      return 'jpeg';
    }
    
    // PNG: 89 50 4E 47 0D 0A 1A 0A
    if (buffer.length >= 8 &&
        buffer.readUInt8(0) === 0x89 &&
        buffer.readUInt8(1) === 0x50 &&
        buffer.readUInt8(2) === 0x4E &&
        buffer.readUInt8(3) === 0x47 &&
        buffer.readUInt8(4) === 0x0D &&
        buffer.readUInt8(5) === 0x0A &&
        buffer.readUInt8(6) === 0x1A &&
        buffer.readUInt8(7) === 0x0A) {
      return 'png';
    }
    
    // GIF: 47 49 46 38
    if (buffer.length >= 4 &&
        buffer.readUInt8(0) === 0x47 &&
        buffer.readUInt8(1) === 0x49 &&
        buffer.readUInt8(2) === 0x46 &&
        buffer.readUInt8(3) === 0x38) {
      return 'gif';
    }
    
    return 'unknown';
  };
  
  // 综合验证逻辑
  validate: async (filePath: string): Promise<ValidationResult> => {
    const ext = path.extname(filePath).toLowerCase();
    
    // 1. 扩展名检查
    const supportedExts = Object.values(this.extensions).flat();
    if (!supportedExts.includes(ext)) {
      return {
        valid: false,
        error: 'UNSUPPORTED_FORMAT',
        message: `不支持的格式：${ext}, 仅支持 ${this.supportedFormats.join(', ')}`,
        suggestion: `请转换为 .jpg, .png 或 .gif 格式`
      };
    }
    
    // 2. 读取文件头验证
    const buffer = await fs.readFile(filePath);
    const detectedFormat = this.validateByMagicNumber(buffer);
    
    if (detectedFormat === 'unknown') {
      return {
        valid: false,
        error: 'INVALID_FILE_HEADER',
        message: '文件头签名无效，可能不是合法图片文件',
        suggestion: '请确认文件未损坏或被篡改'
      };
    }
    
    // 3. 扩展名与内容一致性检查
    const expectedFormat = this.getFormatFromExt(ext);
    if (expectedFormat !== detectedFormat) {
      return {
        valid: false,
        error: 'EXTENSION_MISMATCH',
        message: `文件扩展名 (${ext}) 与实际格式 (${detectedFormat}) 不一致`,
        suggestion: '请使用正确的扩展名或将文件转换为正确格式'
      };
    }
    
    return {
      valid: true,
      format: detectedFormat,
      mimeType: this.mimeTypes[detectedFormat]
    };
  };
  
  // 错误提示文案
  getErrorMessage: (error: Validation Error): string => {
    const messages = {
      UNSUPPORTED_FORMAT: '不支持的图片格式',
      INVALID_FILE_HEADER: '无效的图片文件头',
      EXTENSION_MISMATCH: '文件扩展名与实际格式不匹配'
    };
    
    return messages[error.error];
  }
}

interface ValidationResult {
  valid: boolean;
  format?: 'jpeg' | 'png' | 'gif';
  mimeType?: string;
  error?: string;
  message?: string;
  suggestion?: string;
}
```

```typescript
interface ImageSizeValidator {
  maxSizeBytes: 5 * 1024 * 1024;  // 5MB
  
  validate: async (filePath: string): Promise<ValidationResult> => {
    const stats = await fs.stat(filePath);
    const sizeInMB = stats.size / (1024 * 1024);
    
    if (sizeInMB > this.maxSizeBytes) {
      return {
        valid: false,
        error: 'FILE_TOO_LARGE',
        message: `图片过大：${sizeInMB.toFixed(2)}MB > 5MB`,
        currentSize: stats.size,
        maxSize: this.maxSizeBytes,
        suggestion: `建议压缩到 5MB 以下，或使用在线工具：https://tinypng.com`
      };
    }
    
    return {
      valid: true,
      size: stats.size,
      sizeInMB: sizeInMB.toFixed(2)
    };
  };
  
  // 压缩阈值 (超过此值建议压缩)
  compressionThresholdMB: 2;
  
  shouldCompress: (sizeInMB: number): boolean => {
    return sizeInMB > this.compressionThresholdMB;
  };
}
```

```typescript
interface ImageDimensionAnalyzer {
  // Console 推荐的尺寸
  recommendedSizes = {
    landscape: {
      width: 1280,
      height: 720,
      ratio: 16 / 9,
      aspectLabel: '16:9 (横图)'
    },
    portrait: {
      width: 720,
      height: 1280,
      ratio: 9 / 16,
      aspectLabel: '9:16 (竖图)'
    },
    square: {
      width: 1080,
      height: 1080,
      ratio: 1,
      aspectLabel: '1:1 (正方形)'
    }
  };
  
  // 检测图片维度
  analyzeDimensions: async (imageUrl: string): Promise<DimensionInfo> => {
    const image = sharp(imageUrl);
    const metadata = await image.metadata();
    
    if (!metadata.width || !metadata.height) {
      throw new Error('无法解析图片维度');
    }
    
    const width = metadata.width;
    const height = metadata.height;
    const ratio = width / height;
    
    // 计算最接近的推荐比例
    const closestRecommendation = this.findClosestRecommendation(ratio);
    
    return {
      width,
      height,
      ratio,
      aspectRatioString: this.ratioToFraction(ratio),
      recommendedSize: closestRecommendation,
      meetsRecommendation: Math.abs(ratio - closestRecommendation.ratio) < 0.05
    };
  };
  
  // 比例转分数表示
  ratioToFraction: (ratio: number): string => {
    const tolerance = 0.01;
    const maxDenominator = 100;
    
    let bestFraction = { num: Math.round(ratio), den: 1 };
    let bestDiff = Math.abs(ratio - ratio);
    
    for (let den = 1; den <= maxDenominator; den++) {
      for (let num = Math.floor(ratio * den); num <= Math.ceil(ratio * den); num++) {
        const currentRatio = num / den;
        const diff = Math.abs(ratio - currentRatio);
        
        if (diff < tolerance && diff < bestDiff) {
          bestDiff = diff;
          bestFraction = { num, den };
        }
      }
    }
    
    return `${bestFraction.num}:${bestFraction.den}`;
  };
  
  // 找到最接近的推荐比例
  findClosestRecommendation: (ratio: number): RecommendedSize => {
    const recommendations = Object.values(this.recommendedSizes);
    
    let closest = recommendations[0];
    let minDiff = Math.abs(ratio - closest.ratio);
    
    for (const rec of recommendations) {
      const diff = Math.abs(ratio - rec.ratio);
      if (diff < minDiff) {
        minDiff = diff;
        closest = rec;
      }
    }
    
    return closest;
  };
  
  // TTY 展示信息
  showAnalysis: (info: DimensionInfo): void => {
    console.log(chalk`\n{blue ℹ️} 图片尺寸分析:`);
    console.log(`\n当前尺寸：${info.width} x ${info.height} 像素`);
    console.log(`当前比例：${info.aspectRatioString} (${info.ratio.toFixed(2)})`);
    console.log(`推荐比例：${info.recommendedSize.aspectLabel}`);
    
    if (!info.meetsRecommendation) {
      console.log(chalk`  {yellow ⚠️  非推荐比例，建议调整为 ${info.recommendedSize.aspectLabel} 以获得最佳展示效果}`);
    } else {
      console.log(chalk`  {green ✓} 符合推荐比例`);
    }
  };
}

interface DimensionInfo {
  width: number;
  height: number;
  ratio: number;
  aspectRatioString: string;  // "16:9"
  recommendedSize: RecommendedSize;
  meetsRecommendation: boolean;
}

interface RecommendedSize {
  width: number;
  height: number;
  ratio: number;
  aspectLabel: string;
}
```

#### **Step 5: 图像质量检查 (可选增强)**

```typescript
interface ImageQualityChecker {
  // 模糊度检测 (基于拉普拉斯方差)
  checkBlur: async (imageUrl: string): Promise<BlurInfo> => {
    const image = await sharp(imageUrl).grayscale().toBuffer();
    const grayscale = sharp(image);
    
    const { data } = await grayscale
      .kernel([1, 0, -1, 2, 0, -2, 1, 0, -1])  // 拉普拉斯算子
      .ensureAlpha()
      .toData();
    
    // 计算方差
    let sum = 0;
    const len = data!.length / 4;
    for (let i = 0; i < len; i++) {
      sum += data![i * 4];
    }
    const mean = sum / len;
    
    let variance = 0;
    for (let i = 0; i < len; i++) {
      variance += Math.pow(data![i * 4] - mean, 2);
    }
    variance /= len;
    
    const blurScore = Math.sqrt(variance);
    
    // 判断是否模糊 (阈值可调)
    const isBlurry = blurScore < 50;
    
    return {
      blurScore,
      isBlurry,
      recommendation: isBlurry ? '图片略显模糊，建议更换更清晰的封面' : '图片清晰度正常'
    };
  };
  
  // 亮度检测 (基于平均灰度值)
  checkBrightness: async (imageUrl: string): Promise<BrightnessInfo> => {
    const { data, info } = await sharp(imageUrl).raw().toData();
    
    if (!data || !info) {
      throw new Error('无法解析图片');
    }
    
    let sum = 0;
    const channels = info.channels;
    const pixels = info.width * info.height;
    
    for (let i = 0; i < data.length; i += channels) {
      sum += data[i];  // R 通道
    }
    
    const avgBrightness = (sum / pixels) / 255 * 100;
    
    let status: 'tooDark' | 'normal' | 'tooBright' = 'normal';
    let message = '亮度正常';
    
    if (avgBrightness < 30) {
      status = 'tooDark';
      message = '图片较暗，可能影响可读性';
    } else if (avgBrightness > 80) {
      status = 'tooBright';
      message = '图片过亮，可能损失细节';
    }
    
    return {
      avgBrightness,
      status,
      message
    };
  };
  
  // 综合质量报告
  generateQualityReport: async (imageUrl: string): Promise<QualityReport> => {
    const [blur, brightness] = await Promise.all([
      this.checkBlur(imageUrl),
      this.checkBrightness(imageUrl)
    ]);
    
    return {
      blur,
      brightness,
      overallScore: this.calculateOverallScore(blur, brightness),
      suggestions: this.generateSuggestions(blur, brightness)
    };
  };
  
  // 综合评分计算
  calculateOverallScore: (blur: BlurInfo, brightness: BrightnessInfo): number => {
    let score = 100;
    
    if (blur.isBlurry) score -= 30;
    if (brightness.status === 'tooDark' || brightness.status === 'tooBright') score -= 10;
    
    return Math.max(0, score);
  };
  
  // 生成优化建议
  generateSuggestions: (blur: BlurInfo, brightness: BrightnessInfo): string[] => {
    const suggestions: string[] = [];
    
    if (blur.isBlurry) {
      suggestions.push('替换为更清晰的图片');
    }
    
    if (brightness.status === 'tooDark') {
      suggestions.push('提高图片亮度或改用亮色背景');
    } else if (brightness.status === 'tooBright') {
      suggestions.push('降低图片曝光或改用深色背景');
    }
    
    return suggestions;
  };
}

interface BlurInfo {
  blurScore: number;
  isBlurry: boolean;
  recommendation: string;
}

interface BrightnessInfo {
  avgBrightness: number;
  status: 'tooDark' | 'normal' | 'tooBright';
  message: string;
}

interface QualityReport {
  blur: BlurInfo;
  brightness: BrightnessInfo;
  overallScore: number;
  suggestions: string[];
}
```

```typescript
interface ImageCompressor {
  qualityLevel: 85;  // JPEG 质量等级 85%(平衡质量和体积)
  targetMaxKB: 2048;  // 目标最大 2MB
  
  compress: async (inputPath: string): Promise<CompressedResult> => {
    const inputStats = await fs.stat(inputPath);
    const inputSizeKB = inputStats.size / 1024;
    
    if (inputSizeKB <= this.targetMaxKB) {
      return {
        skipped: true,
        reason: '原始文件已小于目标大小',
        outputPath: inputPath,
        originalSize: inputStats.size,
        compressedSize: inputStats.size
      };
    }
    
    try {
      const outputPath = `${inputPath}.compressed.jpg`;
      
      await sharp(inputPath)
        .jpeg({
          quality: this.qualityLevel,
          progressive: true,  // 渐进式 JPEG
          force: false
        })
        .toFile(outputPath);
      
      const outputStats = await fs.stat(outputPath);
      
      return {
        skipped: false,
        reason: null,
        outputPath,
        originalSize: inputStats.size,
        compressedSize: outputStats.size,
        reductionPercent: ((inputStats.size - outputStats.size) / inputStats.size * 100).toFixed(2)
      };
    } catch (error) {
      throw new Error(`压缩失败：${error.message}`);
    }
  };
  
  // TTY 压缩反馈
  showCompressionResult: (result: CompressedResult): void => {
    if (result.skipped) {
      console.log(chalk`  {dim ℹ️ 无需压缩，原文件已符合要求}`);
      return;
    }
    
    const savingsKB = result.originalSize - result.compressedSize;
    const savingsPercent = parseFloat(result.reductionPercent);
    
    console.log(chalk`  {green ✓} 成功压缩`);
    console.log(`     原始大小：${(result.originalSize / 1024).toFixed(1)}KB`);
    console.log(`     压缩后：{(result.compressedSize / 1024).toFixed(1)}KB`);
    console.log(chalk`     节省：{green ${savingsPercent}% (${savingsKB.toFixed(1)}KB)}`);
    console.log(`     输出：${path.basename(result.outputPath)}`);
  };
}

interface CompressedResult {
  skipped: boolean;
  reason: string | null;
  outputPath: string;
  originalSize: number;
  compressedSize: number;
  reductionPercent: string;
}
```

```typescript
interface ImageUploader {
  // 上传到 Storage
  uploadToStorage: async (
    filePath?: string,
    url?: string
  ): Promise<UploadResult> => {
    
    let buffer: Buffer;
    
    if (filePath) {
      // 本地文件模式
      buffer = await fs.readFile(filePath);
    } else if (url) {
      // URL 引用模式
      const response = await fetch(url);
      if (!response.ok) {
        throw new UserFacingError('URL_FETCH_FAILED', {
          message: '无法下载指定 URL 的图片',
          url
        });
      }
      buffer = Buffer.from(await response.arrayBuffer());
    } else {
      throw new Error('必须提供 filePath 或 url');
    }
    
    // 调用平台 API
    const result = await platform.storage.uploadImage(buffer);
    
    return {
      imageId: result.imageId,
      cdnUrl: result.cdnUrl,
      uploadedAt: new Date().toISOString()
    };
  };
  
  // 重试机制
  uploadWithRetry: async (
    filePath: string,
    maxRetries: number = 3
  ): Promise<UploadResult> => {
    let lastError: Error;
    
    for (let attempt = 1; attempt <= maxRetries; attempt++) {
      try {
        return await this.uploadToStorage(filePath);
      } catch (error) {
        lastError = error as Error;
        
        if (attempt < maxRetries) {
          const delay = 1000 * Math.pow(2, attempt - 1);
          console.log(chalk`{yellow 重试中... (#{attempt}/${maxRetries}), 等待 ${(delay / 1000).toFixed(1)}秒}`);
          await sleep(delay);
        }
      }
    }
    
    throw new UserFacingError('UPLOAD_FAILED', {
      message: '图片上传失败，请检查网络后重试',
      detail: lastError?.message
    });
  };
}

interface UploadResult {
  imageId: string;
  cdnUrl: string;
  uploadedAt: string;
}
```

```typescript
interface CoverCheckpointHandler {
  // 保存时机
  savePoints = [
    'after_local_validation',    // 本地验证通过后立即保存
    'after_upload_complete',     // 上传成功后保存
    'before_session_exit'        // 退出前强制保存
  ];
  
  // checkpoint 数据结构
  structure = {
    coverSource: 'local-file' | 'url-reference',
    originalFilePath?: string,
    validationResults: {
      format: 'valid' | 'invalid',
      size: number,
      dimensions: { width: number; height: number },
      qualityScore: number
    },
    uploadResult?: {
      imageId: string;
      cdnUrl: string;
      uploadedAt: string;
    },
    compressedPath?: string,      // 如经过压缩
    processingTime: number         // ms
  };
  
  // 恢复逻辑
  resume: (checkpoint: CoverCheckpoint): ResumeDecision => {
    // 如果已上传成功，可跳过上传步骤
    if (checkpoint.uploadResult) {
      return {
        allowed: true,
        skipSteps: ['validation', 'upload'],
        preservedData: {
          cdnUrl: checkpoint.uploadResult.cdnUrl,
          imageId: checkpoint.uploadResult.imageId
        }
      };
    }
    
    // 仅做了本地验证，需重新上传
    if (checkpoint.validationResults.format === 'valid') {
      return {
        allowed: true,
        skipSteps: ['validation'],
        requiredSteps: ['upload'],
        restoredInput: checkpoint.coverSource === 'local-file' 
          ? checkpoint.originalFilePath 
          : null
      };
    }
    
    // 验证失败，需重新处理
    return {
      allowed: false,
      reason: 'VALIDATION_FAILED',
      message: '封面验证失败，需重新选择或处理图片'
    };
  };
}
```
