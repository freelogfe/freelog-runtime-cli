## 🔍 **二、核心组件实现**

### **2.1 子条目策略继承规则**

```typescript
interface SubcollectionStrategyRule {
  mode: 'inherit' | 'force' | 'independent';
  
  // 继承模式：子条目的策略将覆盖合集策略
  inherit?: {
    mergeBehavior: 'merge' | 'override';
    priorityFields: string[]; // 哪些字段不继承
  };
  
  // 强制统一模式：所有子条目遵循合集策略
  force?: {
    syncInterval: number; // 分钟，默认 1440(24h)
    lockedFields: string[]; // 锁定字段
    exceptions: {
      resourceId: string;
      exemptedUntil: string; // 解除封锁时间
    }[];
  };
  
  // 独立控制模式：子条目可有自己的策略
  independent?: {
    requireReview: boolean; // 是否要求审核
    allowedTemplates: string[]; // 允许使用的策略模板 ID 列表
  };
}

class SubcollectionStrategyManager {
  async applyRule(collectionId: string, rule: SubcollectionStrategyRule): Promise<ApplyResult> {
    
    if (rule.mode === 'force') {
      // 获取所有子条目
      const subitems = await this.getCollectionSubitems(collectionId);
      
      // 批量更新策略
      const updates = subitems.map(item => ({
        resourceId: item.resourceId,
        strategyTemplateId: collectionStrategy.templateId,
        parameters: collectionStrategy.parameters,
        syncScheduledAt: new Date()
      }));
      
      return await api.collection.batchUpdateSubitemStrategies(updates);
    }
    
    // 其他模式的处理逻辑...
  }
}
```

class StrategyPreviewRenderer {
  render(compiledPolicy: CompiledPolicy, subcollectionRule: SubcollectionStrategyRule) {
    console.log('\n┌─ 策略执行效果预览 ────────────────────┐');
    console.log('│                                       │');
    
    // 测试场景 1: 普通用户下载

      name: '普通用户下载单个子条目',
      userContext: { isVerified: false, isCommercial: false },
      expectedResult: 'ALLOWED'
    }, compiledPolicy);
    
    // 测试场景 2: 公司在商业项目中使用
    this.renderScenario({
      name: '公司在商业项目中使用合集',
      userContext: { isVerified: true, isCommercial: true },
      expectedResult: 'REQUIRES_LICENSE'
    }, compiledPolicy);
    
    console.log('│');
    console.log('│ 📊 子条目策略继承模式：');
    console.log(`│   ${subcollectionRule.mode.replace('-', ' ')}模式`);
    
    if (subcollectionRule.mode === 'force') {
      console.log(`│   • 同步间隔：${subcollectionRule.force?.syncInterval || 1440}分钟`);
      console.log(`│   • 锁定字段：${subcollectionRule.force?.lockedFields.join(', ')}`);
    }
    
    console.log('│');
    console.log('└───────────────────────────────────────┘\n');
  }
  
  private renderScenario(scenario: TestScenario, policy: CompiledPolicy) {
    const result = this.evaluatePolicy(scenario, policy);
    
    const statusIcon = result.status === 'ALLOWED' ? '✅' : 
                      result.status === 'REQUIRES_LICENSE' ? '⚠️' : '❌';
    
    console.log(`│ ${scenario.name}:`);
    console.log(`│   ${statusIcon} ${result.status}`);
    
    if (result.conditions.length > 0) {
      result.conditions.forEach(cond => {
        console.log(`│   条件：${cond}`);
      });
    }
    
    if (result.limits.length > 0) {
      result.limits.forEach(limit => {
        console.log(`│   限制：${limit}`);
      });
    }
    
    console.log('│');
  }
}

## 🚨 **三、异常分支处理**

### **3.1 错误码清单**

| 错误码 | 触发条件 | 用户提示 | 修复建议 |
|-------|---------|---------|---------|
| POLICY_COMPILE_ERROR | 策略编译失败 | "策略编译失败：{reason}" | 检查参数合法性 |
| SUBITEM_SYNC_FAILED | 子条目同步失败 | "{count}个条目同步失败" | 查看详细报告后重试 |
| FORCE_MODE_CONFLICT | 强制模式与现有例外冲突 | "存在无法删除的例外规则" | 先清理例外再启用强制模式 |
| INDEPENDENT_REVIEW_REQUIRED | 独立模式需要审核 | "策略变更需要通过审核" | 提交审核或改用其他模式 |
| STRATEGY_VERSION_MISMATCH | 策略版本不兼容 | "策略模板版本不支持" | 使用更新的模板 |

### **3.2 子条目同步失败处理**

```typescript
async function handleSubitemSyncFailure(
  failedItems: FailedSubitem[],
  onRetry: () => Promise<void>
): Promise<HandleResult> {
  
  if (failedItems.length > 10) {
    warn(`有${failedItems.length}个条目同步失败，已暂停批量更新`);
    
    console.log('\n┌─ 失败详情 ──────────────────────────┐');
    console.log('│                                     │');
    
    failedItems.slice(0, 5).forEach(item => {
      console.log(`│ ✗ ${item.title} - ${item.errorCode}`);
    });
    
    console.log('│ ... (' + (failedItems.length - 5) + ' more)');
    console.log('│');
    console.log('│ [查看全部] V | [重试失败项] R | [放弃] A│');
    console.log('└─────────────────────────────────────┘\n');
    
    return { action: 'show_failed_details' };
  }
  
  // 少量失败，显示警告但继续
  warn(`有 ${failedItems.length} 个条目同步失败:`);
  failedItems.forEach(item => {
    console.log(`  ⚠️  ${item.title}: ${item.errorCode}`);
  });
  
  return { action: 'continue_with_warnings' };
}
```
