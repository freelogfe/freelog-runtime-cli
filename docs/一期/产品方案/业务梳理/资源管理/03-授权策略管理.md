## 🔍 **二、核心组件实现**

### **2.1 策略模板加载器**

```typescript
interface StrategyTemplateLoader {
  async loadTemplates(): Promise<PolicyTemplate[]>;
  async getTemplateById(templateId: string): Promise<PolicyTemplate>;
  async getBuiltInTemplates(): Promise<PolicyTemplate[]>;
}

class BuiltInTemplateCache implements StrategyTemplateLoader {
  private cache: Map<string, PolicyTemplate> = new Map();
  private lastUpdated: Date = new Date();
  private readonly TTL = 30 * 60 * 1000; // 30 分钟缓存
  
  async loadTemplates(): Promise<PolicyTemplate[]> {
    // 从本地缓存或远端加载
    if (this.cache.size > 0 && this.isCacheValid()) {
      return Array.from(this.cache.values());
    }
    
    const templates = await api.policy.getTemplates();
    
    templates.forEach(t => {
      this.cache.set(t.id, t);
    });
    
    this.lastUpdated = new Date();
    return templates;
  }
  
  private isCacheValid(): boolean {
    return Date.now() - this.lastUpdated.getTime() < this.TTL;
  }
}
```

### **2.2 策略编译器**

```typescript
interface PolicyCompiler {
  compile(templateId: string, parameters: Record<string, any>): CompiledPolicy;
  validate(compiledPolicy: CompiledPolicy): ValidationResult;
  toJSON(compiledPolicy: CompiledPolicy): PolicyJSON;
}

class DynamicPolicyCompiler implements PolicyCompiler {
  async compile(templateId: string, parameters: Record<string, any>): Promise<CompiledPolicy> {
    const template = await this.templateLoader.getTemplateById(templateId);
    
    // 1. 数据验证
    const validatedParams = template.formSchema.parse(parameters);
    
    // 2. 动态表单渲染器生成策略逻辑
    const policyLogic = DynamicFormRenderer.generatePolicyLogic(
      template.formSchema,
      validatedParams
    );
    
    // 3. 编译为可执行字节码
    const bytecode = PolicyEngine.compile({
      type: template.type,
      logic: policyLogic
    });
    
    // 4. 计算 checksum
    const checksum = this.computeChecksum(bytecode);
    
    return {
      version: '1.0',
      type: template.type,
      bytecode,
      checksum,
      parameters: validatedParams,
      compiledAt: new Date().toISOString()
    };
  }
  
  private computeChecksum(bytecode: Buffer): string {
    return crypto.createHash('sha256').update(bytecode).digest('hex');
  }
}
```

## 🚨 **三、异常分支处理**

### **3.1 错误码清单**

| 错误码 | 触发条件 | 用户提示 | 修复建议 |
|-------|---------|---------|---------|
| POLICY_COMPILE_ERROR | 策略编译失败 | "策略编译失败：{reason}" | 检查参数是否合法 |
| POLICY_VALIDATION_FAILED | 策略验证失败 | "策略不符合平台规范" | 修改策略逻辑 |
| TEMPLATE_NOT_FOUND | 模板不存在 | "所选策略模板不存在" | 刷新模板列表后重试 |
| BYTECODE_SIZE_EXCEEDED | 字节码过大 | "策略字节码超过限制 (50KB)" | 简化策略逻辑 |
| PERMISSION_CONFLICT | 权限冲突 | "检测到冲突的权限定义" | 移除重复权限 |

### **3.2 策略编译错误处理**

```typescript
async function handlePolicyCompileError(
  error: CompileError,
  onRetry: () => Promise<void>
): Promise<HandleResult> {
  
  const errorCode = extractErrorCode(error);
  
  switch (errorCode) {
    case 'FORM_SCHEMA_VALIDATION':
      console.log(ui.error('策略参数验证失败'));
      console.log(ui.details(error.message));
      return {
        action: 'show-form',
        focusField: error.invalidField
      };
      
    case 'PERMISSION_CONFLICT':
      console.log(ui.warning('检测到权限冲突'));
      console.log(ui.list(error.conflictingPermissions));
      return {
        action: 'merge-suggestions',
        suggestions: generateMergeSuggestions(error)
      };
      
    case 'BYTECODE_OVERFLOW':
      console.log(ui.error('策略过于复杂'));
      console.log('建议：简化权限逻辑或使用预设模板');
      return {
        action: 'suggest-template',
        templates: SIMPLER_TEMPLATES
      };
      
    default:
      console.log(ui.error('策略编译失败'));
      console.log(ui.details(error.stack));
      return {
        action: 'manual-debug',
        enableDebugMode: true
      };
  }
}
```

### **4.2 回滚操作**

```typescript
async function rollbackToPreviousPolicy(
  resourceId: string,
  targetVersion: string
): Promise<RollbackResult> {
  
  // 1. 查找目标版本的策略快照
  const targetPolicy = await getPolicySnapshot(resourceId, targetVersion);
  if (!targetPolicy) {
    throw new ValidationError('POLICY_SNAPSHOT_NOT_FOUND', '找不到该版本的策略快照');
  }
  
  // 2. 执行回滚
  const response = await api.resource.rollbackPolicy({
    resourceId,
    targetPolicy: targetPolicy.checksum
  });
  
  return {
    success: true,
    rolledBackTo: targetVersion,
    newPolicyId: response.newPolicyId,
    effectiveAt: new Date().toISOString()
  };
}
```
