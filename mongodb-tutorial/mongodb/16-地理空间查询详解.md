# 第十六章：地理空间查询详解

## 16.1 地理空间数据概述

### 地理空间数据类型

MongoDB 支持两种地理空间数据格式：

1. **GeoJSON**：标准的地理空间数据格式（推荐）
2. **Legacy Coordinates**：传统坐标对格式

### GeoJSON 格式

```javascript
// Point（点）
{
  location: {
    type: "Point",
    coordinates: [116.3974, 39.9093]  // [经度, 纬度]
  }
}

// LineString（线）
{
  route: {
    type: "LineString",
    coordinates: [
      [116.3974, 39.9093],
      [116.4074, 39.9193],
      [116.4174, 39.9293]
    ]
  }
}

// Polygon（多边形）
{
  area: {
    type: "Polygon",
    coordinates: [[
      [116.3974, 39.9093],
      [116.4074, 39.9093],
      [116.4074, 39.9193],
      [116.3974, 39.9193],
      [116.3974, 39.9093]  // 闭合多边形
    ]]
  }
}
```

## 16.2 地理空间索引

### 2dsphere 索引（推荐）

```javascript
// 创建 2dsphere 索引（球面坐标）
db.places.createIndex({ location: "2dsphere" });

// 支持 GeoJSON 格式
// 适用于：地球表面的地理位置查询
```

### 2d 索引（平面坐标）

```javascript
// 创建 2d 索引（平面坐标）
db.places.createIndex({ location: "2d" });

// 支持传统坐标对格式：[x, y]
// 适用于：平面地图、游戏地图等
```

### 复合索引

```javascript
// 地理空间 + 其他字段
db.places.createIndex({
  location: "2dsphere",
  category: 1,
  rating: -1
});
```

## 16.3 地理空间查询操作符

### $near - 附近查询

```javascript
// 查找附近的点（按距离排序）
db.places.find({
  location: {
    $near: {
      $geometry: {
        type: "Point",
        coordinates: [116.3974, 39.9093]  // 北京天安门
      },
      $maxDistance: 1000,  // 1 公里内（米）
      $minDistance: 100    // 至少 100 米
    }
  }
});

// 使用传统坐标格式
db.places.find({
  location: {
    $near: [116.3974, 39.9093],
    $maxDistance: 1000
  }
});
```

### $geoWithin - 范围内查询

```javascript
// 查找多边形内的点
db.places.find({
  location: {
    $geoWithin: {
      $geometry: {
        type: "Polygon",
        coordinates: [[
          [116.3974, 39.9093],
          [116.4074, 39.9093],
          [116.4074, 39.9193],
          [116.3974, 39.9193],
          [116.3974, 39.9093]
        ]]
      }
    }
  }
});

// 使用圆形范围
db.places.find({
  location: {
    $geoWithin: {
      $centerSphere: [
        [116.3974, 39.9093],  // 中心点
        1 / 6378.1  // 半径（1 公里，6378.1 是地球半径，单位：公里）
      ]
    }
  }
});

// 使用矩形范围
db.places.find({
  location: {
    $geoWithin: {
      $box: [
        [116.3874, 39.8993],  // 左下角
        [116.4074, 39.9193]   // 右上角
      ]
    }
  }
});
```

### $geoIntersects - 相交查询

```javascript
// 查找与指定几何图形相交的文档
db.areas.find({
  boundary: {
    $geoIntersects: {
      $geometry: {
        type: "Point",
        coordinates: [116.3974, 39.9093]
      }
    }
  }
});

// 查找与线相交的区域
db.areas.find({
  boundary: {
    $geoIntersects: {
      $geometry: {
        type: "LineString",
        coordinates: [
          [116.3974, 39.9093],
          [116.4074, 39.9193]
        ]
      }
    }
  }
});
```

### $nearSphere - 球面附近查询

```javascript
// 使用球面距离计算（更准确）
db.places.find({
  location: {
    $nearSphere: {
      $geometry: {
        type: "Point",
        coordinates: [116.3974, 39.9093]
      },
      $maxDistance: 1000  // 1 公里
    }
  }
});
```

## 16.4 地理空间聚合操作

### $geoNear - 地理空间聚合

```javascript
// 地理空间聚合查询
db.places.aggregate([
  {
    $geoNear: {
      near: {
        type: "Point",
        coordinates: [116.3974, 39.9093]
      },
      distanceField: "distance",  // 距离字段名
      spherical: true,              // 使用球面距离
      maxDistance: 1000,            // 最大距离（米）
      query: { category: "restaurant" },  // 额外查询条件
      includeLocs: "location",      // 包含位置信息
      num: 10                       // 返回数量
    }
  },
  {
    $project: {
      name: 1,
      category: 1,
      distance: { $divide: ["$distance", 1000] },  // 转换为公里
      location: 1
    }
  }
]);
```

### 计算距离

```javascript
// 使用聚合管道计算距离
db.places.aggregate([
  {
    $geoNear: {
      near: {
        type: "Point",
        coordinates: [116.3974, 39.9093]
      },
      distanceField: "distance",
      spherical: true
    }
  },
  {
    $addFields: {
      distanceKm: {
        $divide: ["$distance", 1000]
      }
    }
  },
  {
    $match: {
      distanceKm: { $lte: 5 }  // 5 公里内
    }
  },
  {
    $sort: { distance: 1 }
  }
]);
```

## 16.5 实际应用场景

### 场景 1：附近商家搜索

```typescript
// 查找附近的餐厅
async function findNearbyRestaurants(
  longitude: number,
  latitude: number,
  maxDistance: number = 1000
) {
  return await Place.find({
    location: {
      $near: {
        $geometry: {
          type: "Point",
          coordinates: [longitude, latitude]
        },
        $maxDistance: maxDistance
      }
    },
    category: "restaurant"
  })
  .limit(20);
}
```

### 场景 2：配送范围检查

```typescript
// 检查地址是否在配送范围内
async function isInDeliveryArea(
  longitude: number,
  latitude: number,
  restaurantId: string
) {
  const restaurant = await Restaurant.findById(restaurantId);
  
  return await Restaurant.findOne({
    _id: restaurantId,
    deliveryArea: {
      $geoIntersects: {
        $geometry: {
          type: "Point",
          coordinates: [longitude, latitude]
        }
      }
    }
  });
}
```

## 16.6 Mongoose 中的地理空间

### 模型定义

```typescript
import mongoose from 'mongoose';

const placeSchema = new mongoose.Schema({
  name: String,
  location: {
    type: {
      type: String,
      enum: ['Point'],
      required: true
    },
    coordinates: {
      type: [Number],
      required: true,
      validate: {
        validator: (coords: number[]) => {
          return coords.length === 2 && 
                 coords[0] >= -180 && coords[0] <= 180 &&
                 coords[1] >= -90 && coords[1] <= 90;
        },
        message: '坐标格式不正确'
      }
    }
  },
  category: String
});

// 创建地理空间索引
placeSchema.index({ location: '2dsphere' });

const Place = mongoose.model('Place', placeSchema);
```

### 查询示例

```typescript
// 查找附近的商家
async function findNearby(longitude: number, latitude: number) {
  return await Place.find({
    location: {
      $near: {
        $geometry: {
          type: 'Point',
          coordinates: [longitude, latitude]
        },
        $maxDistance: 1000
      }
    }
  });
}
```

## 16.7 坐标系统

### WGS84（世界大地测量系统）

```javascript
// MongoDB 默认使用 WGS84
// 经度范围：-180 到 180
// 纬度范围：-90 到 90

{
  location: {
    type: "Point",
    coordinates: [116.3974, 39.9093]  // [经度, 纬度]
  }
}
```

### 坐标转换

```typescript
// GCJ-02（中国坐标系）转 WGS84
function gcj02ToWgs84(lng: number, lat: number): [number, number] {
  // 转换算法（简化版）
  const dLat = transformLat(lng - 105.0, lat - 35.0);
  const dLng = transformLng(lng - 105.0, lat - 35.0);
  const radLat = lat / 180.0 * Math.PI;
  let magic = Math.sin(radLat);
  magic = 1 - 0.00669342162296594323 * magic * magic;
  const sqrtMagic = Math.sqrt(magic);
  dLat = (dLat * 180.0) / ((6378245.0 * (1 - 0.00669342162296594323)) / (magic * sqrtMagic) * Math.PI);
  dLng = (dLng * 180.0) / (6378245.0 / sqrtMagic * Math.cos(radLat) * Math.PI);
  return [lng - dLng, lat - dLat];
}
```

## 16.9 常见问题和注意事项

### 问题 1：坐标顺序

```javascript
// ❌ 错误：坐标顺序错误（应该是 [经度, 纬度]）
{
  location: {
    type: "Point",
    coordinates: [39.9093, 116.3974]  // 错误：纬度在前
  }
}

// ✅ 正确：坐标顺序正确
{
  location: {
    type: "Point",
    coordinates: [116.3974, 39.9093]  // 正确：经度在前
  }
}
```

### 问题 2：多边形闭合

```javascript
// ❌ 错误：多边形未闭合
{
  area: {
    type: "Polygon",
    coordinates: [[
      [116.3974, 39.9093],
      [116.4074, 39.9093],
      [116.4074, 39.9193],
      [116.3974, 39.9193]
      // 缺少最后一个点
    ]]
  }
}

// ✅ 正确：多边形闭合
{
  area: {
    type: "Polygon",
    coordinates: [[
      [116.3974, 39.9093],
      [116.4074, 39.9093],
      [116.4074, 39.9193],
      [116.3974, 39.9193],
      [116.3974, 39.9093]  // 闭合多边形
    ]]
  }
}
```

### 问题 3：距离单位

```javascript
// 注意：$maxDistance 的单位是米
db.places.find({
  location: {
    $near: {
      $geometry: {
        type: "Point",
        coordinates: [116.3974, 39.9093]
      },
      $maxDistance: 1000  // 1000 米 = 1 公里
    }
  }
});
```
