# 第十三章：MongoDB 安全认证与备份恢复

## 13.1 MongoDB 安全概述

### 安全最佳实践

1. **启用认证**：使用用户名和密码
2. **网络隔离**：限制网络访问
3. **加密传输**：使用 TLS/SSL
4. **角色权限**：最小权限原则
5. **定期审计**：监控和日志记录
6. **数据备份**：定期备份数据

## 13.2 用户认证

### 创建管理员用户

```javascript
// 1. 启动 MongoDB（无认证模式）
mongod --dbpath /data/db

// 2. 连接到 MongoDB
mongosh

// 3. 切换到 admin 数据库
use admin

// 4. 创建管理员用户
db.createUser({
  user: "admin",
  pwd: "securePassword123",
  roles: [
    { role: "userAdminAnyDatabase", db: "admin" },
    { role: "dbAdminAnyDatabase", db: "admin" },
    { role: "readWriteAnyDatabase", db: "admin" },
    { role: "clusterAdmin", db: "admin" }
  ]
})
```

### 创建应用用户

```javascript
// 切换到应用数据库
use myapp

// 创建应用用户（只读）
db.createUser({
  user: "readonly",
  pwd: "readonlyPassword",
  roles: [{ role: "read", db: "myapp" }]
})

// 创建应用用户（读写）
db.createUser({
  user: "appuser",
  pwd: "appPassword",
  roles: [
    { role: "readWrite", db: "myapp" },
    { role: "read", db: "config" }
  ]
})
```

### 内置角色说明

| 角色 | 权限 |
|------|------|
| `read` | 只读权限 |
| `readWrite` | 读写权限 |
| `dbAdmin` | 数据库管理权限 |
| `userAdmin` | 用户管理权限 |
| `clusterAdmin` | 集群管理权限 |
| `readAnyDatabase` | 读取所有数据库 |
| `readWriteAnyDatabase` | 读写所有数据库 |

### 启用认证

```bash
# 1. 编辑 mongod.cfg
security:
  authorization: enabled

# 2. 重启 MongoDB
net stop MongoDB
net start MongoDB

# 3. 使用认证连接
mongosh "mongodb://admin:securePassword123@localhost:27017/admin"
```

### 应用中使用认证

```typescript
// Mongoose
mongoose.connect('mongodb://appuser:appPassword@localhost:27017/myapp', {
  authSource: 'myapp'
});

// 原生驱动
const client = new MongoClient('mongodb://appuser:appPassword@localhost:27017/myapp', {
  authSource: 'myapp'
});
```

## 13.3 TLS/SSL 加密

### 生成证书

```bash
# 1. 创建证书目录
mkdir -p /etc/mongodb/ssl

# 2. 生成 CA 证书
openssl req -new -x509 -days 365 -keyout ca.key -out ca.crt -subj "/CN=MongoDB-CA"

# 3. 生成服务器证书
openssl req -new -keyout server.key -out server.csr -subj "/CN=mongodb-server"
openssl x509 -req -days 365 -in server.csr -CA ca.crt -CAkey ca.key -CAcreateserial -out server.crt

# 4. 合并证书和密钥
cat server.crt server.key > server.pem
```

### 配置 MongoDB TLS

```yaml
# mongod.cfg
net:
  port: 27017
  ssl:
    mode: requireSSL
    PEMKeyFile: /etc/mongodb/ssl/server.pem
    CAFile: /etc/mongodb/ssl/ca.crt
    allowConnectionsWithoutCertificates: false
```

### 客户端连接

```typescript
// 使用 TLS 连接
mongoose.connect('mongodb://localhost:27017/myapp', {
  ssl: true,
  sslValidate: true,
  sslCA: fs.readFileSync('/path/to/ca.crt'),
  sslCert: fs.readFileSync('/path/to/client.crt'),
  sslKey: fs.readFileSync('/path/to/client.key'),
});
```

## 13.5 数据备份

### mongodump - 备份数据库

```bash
# 备份整个数据库
mongodump --uri="mongodb://localhost:27017/myapp" --out=/backup/myapp

# 备份特定集合
mongodump --uri="mongodb://localhost:27017/myapp" --collection=users --out=/backup/users

# 带认证备份
mongodump --uri="mongodb://admin:password@localhost:27017/myapp" --out=/backup/myapp

# 压缩备份
mongodump --uri="mongodb://localhost:27017/myapp" --out=/backup/myapp --gzip

# 备份到远程服务器
mongodump --uri="mongodb://localhost:27017/myapp" --out=/backup/myapp --archive | ssh user@remote "cat > /backup/myapp.archive"
```

### 定时备份脚本

```bash
#!/bin/bash
# backup.sh

BACKUP_DIR="/backup/mongodb"
DATE=$(date +%Y%m%d_%H%M%S)
DB_NAME="myapp"
MONGODB_URI="mongodb://localhost:27017/$DB_NAME"

# 创建备份目录
mkdir -p $BACKUP_DIR

# 执行备份
mongodump --uri="$MONGODB_URI" --out="$BACKUP_DIR/$DATE" --gzip

# 删除 7 天前的备份
find $BACKUP_DIR -type d -mtime +7 -exec rm -rf {} \;

echo "Backup completed: $BACKUP_DIR/$DATE"
```

```powershell
# backup.ps1 (Windows)

$BackupDir = "D:\backup\mongodb"
$Date = Get-Date -Format "yyyyMMdd_HHmmss"
$DbName = "myapp"
$MongoUri = "mongodb://localhost:27017/$DbName"

# 创建备份目录
New-Item -ItemType Directory -Force -Path $BackupDir

# 执行备份
mongodump --uri="$MongoUri" --out="$BackupDir\$Date" --gzip

# 删除 7 天前的备份
Get-ChildItem -Path $BackupDir -Directory | Where-Object {
    $_.LastWriteTime -lt (Get-Date).AddDays(-7)
} | Remove-Item -Recurse -Force

Write-Host "Backup completed: $BackupDir\$Date"
```

## 13.6 数据恢复

### mongorestore - 恢复数据库

```bash
# 恢复整个数据库
mongorestore --uri="mongodb://localhost:27017/myapp" /backup/myapp

# 恢复特定集合
mongorestore --uri="mongodb://localhost:27017/myapp" --collection=users /backup/users/users.bson

# 从压缩文件恢复
mongorestore --uri="mongodb://localhost:27017/myapp" --gzip /backup/myapp

# 恢复时删除现有数据
mongorestore --uri="mongodb://localhost:27017/myapp" --drop /backup/myapp
```

### 增量备份和恢复

```bash
# 1. 使用 oplog 进行增量备份
mongodump --uri="mongodb://localhost:27017/myapp" --oplog --out=/backup/incremental

# 2. 恢复时应用 oplog
mongorestore --uri="mongodb://localhost:27017/myapp" --oplogReplay /backup/incremental
```

## 13.7 副本集备份

### 从副本集备份

```bash
# 从主节点备份
mongodump --uri="mongodb://localhost:27017/myapp?replicaSet=rs0" --out=/backup/myapp

# 从从节点备份（减少主节点压力）
mongodump --uri="mongodb://secondary:27017/myapp?replicaSet=rs0&readPreference=secondary" --out=/backup/myapp
```

### 副本集恢复

```bash
# 1. 停止从节点
# 2. 恢复数据到从节点
mongorestore --uri="mongodb://secondary:27017/myapp" /backup/myapp

# 3. 重启从节点，自动同步
```

## 13.8 自动化备份方案

### Node.js 备份脚本

```typescript
// scripts/backup.ts
import { exec } from 'child_process';
import { promisify } from 'util';
import * as fs from 'fs';
import * as path from 'path';

const execAsync = promisify(exec);

interface BackupOptions {
  uri: string;
  outputDir: string;
  dbName: string;
  retentionDays?: number;
}

async function backup(options: BackupOptions) {
  const { uri, outputDir, dbName, retentionDays = 7 } = options;
  const date = new Date().toISOString().replace(/[:.]/g, '-');
  const backupPath = path.join(outputDir, `${dbName}-${date}`);

  // 创建备份目录
  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
  }

  try {
    // 执行备份
    console.log(`开始备份数据库: ${dbName}`);
    await execAsync(
      `mongodump --uri="${uri}" --out="${backupPath}" --gzip`
    );
    console.log(`备份完成: ${backupPath}`);

    // 清理旧备份
    if (retentionDays > 0) {
      const cutoffDate = new Date();
      cutoffDate.setDate(cutoffDate.getDate() - retentionDays);

      const backups = fs.readdirSync(outputDir);
      for (const backup of backups) {
        const backupPath = path.join(outputDir, backup);
        const stats = fs.statSync(backupPath);
        if (stats.mtime < cutoffDate) {
          fs.rmSync(backupPath, { recursive: true });
          console.log(`删除旧备份: ${backup}`);
        }
      }
    }

    return backupPath;
  } catch (error) {
    console.error('备份失败:', error);
    throw error;
  }
}

// 使用
backup({
  uri: 'mongodb://localhost:27017/myapp',
  outputDir: './backups',
  dbName: 'myapp',
  retentionDays: 7
});
```

### 定时任务

```typescript
// 使用 node-cron
import * as cron from 'node-cron';

// 每天凌晨 2 点备份
cron.schedule('0 2 * * *', async () => {
  console.log('执行定时备份...');

    uri: process.env.DATABASE_URL!,
    outputDir: './backups',
    dbName: 'myapp',
    retentionDays: 7
  });
});
```

## 13.9 监控和审计

### 启用审计日志

```yaml
# mongod.cfg (企业版)
auditLog:
  destination: file
  format: JSON
  path: /var/log/mongodb/audit.json
  filter: '{ atype: { $in: [ "authenticate", "createUser", "dropUser" ] } }'
```

### 应用层审计

```typescript
// 记录所有数据库操作
import { Schema } from 'mongoose';

const AuditLogSchema = new Schema({
  operation: String,
  collection: String,
  userId: String,
  details: Schema.Types.Mixed,
  timestamp: { type: Date, default: Date.now }
});

const AuditLog = mongoose.model('AuditLog', AuditLogSchema);

// 中间件记录操作
async function logOperation(operation: string, collection: string, details: any) {
  await AuditLog.create({
    operation,
    collection,
    userId: details.userId,
    details,
    timestamp: new Date()
  });
}
```

## 13.10 安全最佳实践总结

### 1. 认证和授权

```typescript
// ✅ 好：使用认证连接
mongoose.connect('mongodb://user:password@localhost:27017/db', {
  authSource: 'admin'
});

// ❌ 差：无认证连接
mongoose.connect('mongodb://localhost:27017/db');
```

### 2. 最小权限原则

```javascript
// ✅ 好：只授予必要权限
db.createUser({
  user: "appuser",
  pwd: "password",
  roles: [{ role: "readWrite", db: "myapp" }]  // 只访问应用数据库
});

// ❌ 差：授予过多权限
db.createUser({
  user: "appuser",
  pwd: "password",
  roles: [{ role: "root", db: "admin" }]  // 管理员权限
});
```

### 3. 加密传输

```typescript
// ✅ 好：使用 TLS
mongoose.connect(uri, {
  ssl: true,
  sslValidate: true
});
```

### 4. 定期备份

```bash
# ✅ 好：自动化备份
0 2 * * * /path/to/backup.sh
```

### 5. 监控和日志

```typescript
// ✅ 好：记录重要操作
await AuditLog.create({
  operation: 'delete',
  collection: 'users',
  userId: currentUser.id,
  details: { deletedUserId: userId }
});
```
