# 第九章：MongoDB 更新操作详解

## 9.1 更新操作概述

MongoDB 提供了多种更新方法，每种方法适用于不同的场景。

### 更新方法对比

| 方法 | 说明 | 适用场景 |
|------|------|----------|
| `updateOne()` | 更新单个文档 | 精确更新一个文档 |
| `updateMany()` | 更新多个文档 | 批量更新符合条件的文档 |
| `replaceOne()` | 替换整个文档 | 完全替换文档内容 |
| `findOneAndUpdate()` | 查找并更新 | 需要返回更新后的文档 |
| `findOneAndReplace()` | 查找并替换 | 需要返回替换后的文档 |
| `findOneAndDelete()` | 查找并删除 | 需要返回删除的文档 |

## 9.2 更新操作符详解

### $set - 设置字段值

```javascript
// 设置单个字段
db.users.updateOne(
  { name: "张三" },
  { $set: { age: 26 } }
)

// 设置多个字段
db.users.updateOne(
  { name: "张三" },
  { $set: { 
    age: 26,
    email: "zhangsan@example.com",
    status: "active"
  }}
)

// 设置嵌套字段
db.users.updateOne(
  { name: "张三" },
  { $set: { "address.city": "上海" } }
)

// 设置数组元素
db.users.updateOne(
  { name: "张三" },
  { $set: { "hobbies.0": "编程" } }
)
```

### $unset - 删除字段

```javascript
// 删除单个字段
db.users.updateOne(
  { name: "张三" },
  { $unset: { age: "" } }
)

// 删除多个字段
db.users.updateOne(
  { name: "张三" },
  { $unset: { 
    age: "",
    email: "",
    address: ""
  }}
)

// 删除嵌套字段
db.users.updateOne(
  { name: "张三" },
  { $unset: { "address.city": "" } }
)
```

### $inc - 增加/减少数值

```javascript
// 增加数值
db.users.updateOne(
  { name: "张三" },
  { $inc: { score: 10 } }
)

// 减少数值（使用负数）
db.users.updateOne(
  { name: "张三" },
  { $inc: { score: -5 } }
)

// 同时增加多个字段
db.users.updateOne(
  { name: "张三" },
  { $inc: { 
    score: 10,
    level: 1,
    experience: 100
  }}
)
```

### $mul - 乘法运算

```javascript
// 将字段值乘以指定数字
db.products.updateOne(
  { name: "商品A" },
  { $mul: { price: 1.1 } }  // 价格增加 10%
)

// 减少价格（乘以小于1的数）
db.products.updateOne(
  { name: "商品A" },
  { $mul: { price: 0.9 } }  // 价格减少 10%
)
```

### $min / $max - 最小值/最大值

```javascript
// 只有当新值小于当前值时才更新
db.users.updateOne(
  { name: "张三" },
  { $min: { bestScore: 95 } }  // 如果 bestScore > 95，则不更新
)

// 只有当新值大于当前值时才更新
db.users.updateOne(
  { name: "张三" },
  { $max: { bestScore: 95 } }  // 如果 bestScore < 95，则更新为 95
)
```

### $currentDate - 设置当前日期

```javascript
// 设置为当前日期时间
db.users.updateOne(
  { name: "张三" },
  { $currentDate: { lastLogin: true } }
)

// 设置为 Date 类型
db.users.updateOne(
  { name: "张三" },
  { $currentDate: { 
    lastLogin: { $type: "date" }
  }}
)

// 设置为时间戳
db.users.updateOne(
  { name: "张三" },
  { $currentDate: { 
    lastLogin: { $type: "timestamp" }
  }}
)
```

### $rename - 重命名字段

```javascript
// 重命名单个字段
db.users.updateMany(
  {},
  { $rename: { "oldField": "newField" } }
)

// 重命名嵌套字段
db.users.updateMany(
  {},
  { $rename: { "address.city": "address.cityName" } }
)

// 重命名多个字段
db.users.updateMany(
  {},
  { $rename: { 
    "oldField1": "newField1",
    "oldField2": "newField2"
  }}
)

// 注意：$rename 不能用于数组元素
// ❌ 错误：不能重命名数组元素
// { $rename: { "hobbies.0": "hobbies.first" } }
```

### $bit - 位运算操作

`$bit` 操作符用于对整数字段进行位运算（AND、OR、XOR）。

```javascript
// AND 运算（按位与）
db.users.updateOne(
  { name: "张三" },
  { $bit: { flags: { and: 7 } } }  // flags = flags & 7
)

// OR 运算（按位或）
db.users.updateOne(
  { name: "张三" },
  { $bit: { flags: { or: 3 } } }  // flags = flags | 3
)

// XOR 运算（按位异或）
db.users.updateOne(
  { name: "张三" },
  { $bit: { flags: { xor: 5 } } }  // flags = flags ^ 5
)

// 实际应用：权限管理
// flags: 1=读, 2=写, 4=执行
db.users.updateOne(
  { name: "张三" },
  { $bit: { permissions: { or: 3 } } }  // 添加读和写权限（1 | 2 = 3）
)

db.users.updateOne(
  { name: "张三" },
  { $bit: { permissions: { and: 6 } } }  // 移除读权限（保留写和执行）
)
```

### $setOnInsert - 仅在插入时设置

`$setOnInsert` 只在执行插入操作时设置字段值，更新操作时不会修改该字段。

```javascript
// 使用 upsert 时，只在插入时设置默认值
db.users.updateOne(
  { email: "zhangsan@example.com" },
  { 
    $set: { lastLogin: new Date() },  // 每次更新都设置
    $setOnInsert: { 
      createdAt: new Date(),           // 只在插入时设置
      status: "active"                 // 只在插入时设置
    }
  },
  { upsert: true }
)

// 第一次执行：插入文档，设置 createdAt 和 status
// 后续执行：只更新 lastLogin，不修改 createdAt 和 status
```

## 9.3 数组更新操作符

### $push - 添加元素到数组

```javascript
// 添加单个元素
db.users.updateOne(
  { name: "张三" },
  { $push: { hobbies: "编程" } }
)

// 添加多个元素
db.users.updateOne(
  { name: "张三" },
  { $push: { 
    hobbies: { 
      $each: ["编程", "阅读", "运动"]
    }
  }}
)

// 限制数组大小（保留最后 N 个）
db.users.updateOne(
  { name: "张三" },
  { $push: { 
    hobbies: { 
      $each: ["新爱好"],
      $slice: 5  // 只保留最后 5 个
    }
  }}
)

// 按位置插入
db.users.updateOne(
  { name: "张三" },
  { $push: { 
    hobbies: { 
      $each: ["新爱好"],
      $position: 0  // 插入到数组开头
    }
  }}
)

// 排序后限制大小
db.users.updateOne(
  { name: "张三" },
  { $push: { 
    scores: { 
      $each: [95],
      $sort: -1,  // 降序排序
      $slice: 10  // 保留前 10 个
    }
  }}
)
```

### $addToSet - 添加唯一元素

```javascript
// 添加元素（如果不存在）
db.users.updateOne(
  { name: "张三" },
  { $addToSet: { hobbies: "编程" } }
)

// 添加多个唯一元素
db.users.updateOne(
  { name: "张三" },
  { $addToSet: { 
    hobbies: { 
      $each: ["编程", "阅读", "运动"]
    }
  }}
)
```

### $pop - 删除数组元素

```javascript
// 删除最后一个元素
db.users.updateOne(
  { name: "张三" },
  { $pop: { hobbies: 1 } }
)

// 删除第一个元素
db.users.updateOne(
  { name: "张三" },
  { $pop: { hobbies: -1 } }
)
```

### $pull - 删除匹配的元素

```javascript
// 删除匹配的元素
db.users.updateOne(
  { name: "张三" },
  { $pull: { hobbies: "编程" } }
)

// 使用条件删除
db.users.updateOne(
  { name: "张三" },
  { $pull: { 
    scores: { $gte: 100 }  // 删除所有大于等于 100 的分数
  }}
)

// 删除嵌套文档
db.users.updateOne(
  { name: "张三" },
  { $pull: { 
    addresses: { city: "北京" }
  }}
)
```

### $pullAll - 删除多个指定元素

```javascript
// 删除多个指定元素
db.users.updateOne(
  { name: "张三" },
  { $pullAll: { 
    hobbies: ["编程", "阅读"]
  }}
)
```

### 数组位置操作符

```javascript
// 更新数组指定位置的元素
db.users.updateOne(
  { name: "张三", "hobbies.0": "旧爱好" },
  { $set: { "hobbies.0": "新爱好" } }
)

// 使用 $ 位置操作符（更新第一个匹配的元素）
db.users.updateOne(
  { name: "张三", hobbies: "旧爱好" },
  { $set: { "hobbies.$": "新爱好" } }
)

// 使用 $[] 更新所有匹配的元素
db.users.updateMany(
  { hobbies: { $exists: true } },
  { $set: { "hobbies.$[]": "更新后的爱好" } }
)

// 使用 $[<identifier>] 条件更新
db.users.updateMany(
  {},
  { $set: { "hobbies.$[hobby]": "更新后的爱好" } },
  { arrayFilters: [{ "hobby": { $regex: /^旧/ } }] }
)
```

## 9.4 更新选项

### upsert - 不存在则插入

```javascript
// 如果文档不存在，则创建新文档
db.users.updateOne(
  { name: "李四" },
  { 
    $set: { 
      name: "李四",
      age: 30,
      email: "lisi@example.com"
    }
  },
  { upsert: true }
)
```

### arrayFilters - 数组过滤条件

```javascript
// 更新数组中满足条件的元素
db.users.updateMany(
  {},
  { 
    $set: { 
      "scores.$[score]": 0
    }
  },
  { 
    arrayFilters: [
      { "score": { $lt: 60 } }  // 将所有小于 60 的分数设为 0
    ]
  }
)

// 多个条件
db.users.updateMany(
  {},
  { 
    $set: { 
      "grades.$[grade].score": 0
    }
  },
  { 
    arrayFilters: [
      { 
        "grade.subject": "数学",
        "grade.score": { $lt: 60 }
      }
    ]
  }
)
```

### returnDocument - 返回文档选项

```javascript
// 返回更新前的文档（默认）
db.users.findOneAndUpdate(
  { name: "张三" },
  { $set: { age: 26 } },
  { returnDocument: "before" }
)

// 返回更新后的文档
db.users.findOneAndUpdate(
  { name: "张三" },
  { $set: { age: 26 } },
  { returnDocument: "after" }
)
```

## 9.5 实际应用场景

### 场景 1：用户积分系统

```javascript
// 增加积分并更新等级
db.users.updateOne(
  { userId: "user123" },
  { 
    $inc: { 
      points: 100,
      totalPoints: 100
    },
    $set: { 
      lastEarnedAt: new Date()
    },
    $max: { 
      level: Math.floor((currentPoints + 100) / 1000)
    }
  }
)
```

### 场景 2：购物车操作

```javascript
// 添加商品到购物车
db.carts.updateOne(
  { userId: "user123" },
  { 
    $push: {
      items: {
        productId: "prod456",
        quantity: 2,
        addedAt: new Date()
      }
    }
  },
  { upsert: true }
)

// 更新购物车中商品数量
db.carts.updateOne(
  { 
    userId: "user123",
    "items.productId": "prod456"
  },
  { 
    $inc: { "items.$.quantity": 1 }
  }
)

// 从购物车删除商品
db.carts.updateOne(
  { userId: "user123" },
  { 
    $pull: { 
      items: { productId: "prod456" }
    }
  }
)
```

### 场景 3：日志记录

```javascript
// 追加日志条目（限制数量）
db.systems.updateOne(
  { systemId: "sys001" },
  { 
    $push: {
      logs: {
        $each: [{
          level: "error",
          message: "系统错误",
          timestamp: new Date()
        }],
        $slice: -100  // 只保留最后 100 条日志
      }
    }
  }
)
```

### 场景 4：投票系统

```javascript
// 添加投票（确保唯一性）
db.polls.updateOne(
  { pollId: "poll123" },
  { 
    $addToSet: { 
      voters: "user123"
    },
    $inc: { 
      "options.option1.votes": 1
    }
  }
)
```

### 场景 5：软删除

```javascript
// 标记为已删除
db.users.updateOne(
  { userId: "user123" },
  { 
    $set: { 
      isDeleted: true,
      deletedAt: new Date()
    },
    $unset: { 
      email: "",
      phone: ""
    }
  }
)
```

## 9.6 TypeScript 中的更新操作

### Mongoose 示例

```typescript
import { Model } from 'mongoose';

// 更新单个文档
await UserModel.updateOne(
  { name: "张三" },
  { $set: { age: 26 } }
);

// 更新多个文档
await UserModel.updateMany(
  { age: { $lt: 18 } },
  { $set: { status: "minor" } }
);

// 查找并更新
const updatedUser = await UserModel.findOneAndUpdate(
  { name: "张三" },
  { $inc: { score: 10 } },
  { 
    returnDocument: "after",
    new: true  // Mongoose 选项
  }
);
```

### Typegoose 示例

```typescript
import { ReturnModelType } from "@typegoose/typegoose";

@Provide()
export class UserService {
  @InjectEntityModel(User)
  userModel: ReturnModelType<typeof User>;

  async updateUserScore(userId: string, points: number) {
    return await this.userModel.updateOne(
      { _id: userId },
      { 
        $inc: { score: points },
        $set: { lastUpdated: new Date() }
      }
    );
  }

  async addHobby(userId: string, hobby: string) {
    return await this.userModel.updateOne(
      { _id: userId },
      { $addToSet: { hobbies: hobby } }
    );
  }
}
```

### Prisma 示例

```typescript
// Prisma 6.x 更新操作
await prisma.user.update({

    age: 26,

```

## 9.7 性能优化建议

### 1. 批量更新

```javascript
// ❌ 差：循环更新
for (const userId of userIds) {
  await db.users.updateOne({ userId }, { $set: { status: "active" } });
}

// ✅ 好：批量更新
await db.users.updateMany(
  { userId: { $in: userIds } },
  { $set: { status: "active" } }
);
```

### 2. 使用索引

```javascript
// 为更新查询字段创建索引
db.users.createIndex({ userId: 1 });
db.users.createIndex({ "items.productId": 1 });
```

### 3. 避免全表更新

```javascript
// ❌ 差：更新所有文档
db.users.updateMany({}, { $set: { newField: "value" } });

// ✅ 好：只更新需要的文档
db.users.updateMany(
  { status: "active" },
  { $set: { newField: "value" } }
);
```

### 4. 使用 upsert 避免先查询

```javascript
// ❌ 差：先查询再更新
const user = await db.users.findOne({ userId });
if (!user) {
  await db.users.insertOne({ userId, ... });
} else {
  await db.users.updateOne({ userId }, { $set: { ... } });
}

// ✅ 好：使用 upsert
await db.users.updateOne(
  { userId },
  { $set: { ... } },
  { upsert: true }
);
```

## 9.8 常见错误和注意事项

### 1. 忘记使用 $set

```javascript
// ❌ 错误：会替换整个文档
db.users.updateOne({ name: "张三" }, { age: 26 });

// ✅ 正确：只更新指定字段
db.users.updateOne({ name: "张三" }, { $set: { age: 26 } });
```

### 2. 数组操作符误用

```javascript
// ❌ 错误：直接赋值会替换整个数组
db.users.updateOne(
  { name: "张三" },
  { hobbies: ["新爱好"] }
);

// ✅ 正确：使用 $push 添加元素
db.users.updateOne(
  { name: "张三" },
  { $push: { hobbies: "新爱好" } }
);
```

### 3. 嵌套字段更新

```javascript
// ❌ 错误：无法更新嵌套字段
db.users.updateOne(
  { name: "张三" },
  { address: { city: "上海" } }
);

// ✅ 正确：使用点号表示法
db.users.updateOne(
  { name: "张三" },
  { $set: { "address.city": "上海" } }
);
```
