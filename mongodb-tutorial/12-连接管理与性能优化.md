# 第十二章：连接管理与性能优化

## 12.1 连接池基础

### 为什么需要连接池？

- ✅ 减少连接开销：复用连接而不是每次创建新连接
- ✅ 提高性能：避免频繁建立和关闭连接
- ✅ 资源管理：限制并发连接数，防止资源耗尽
- ✅ 连接复用：多个请求共享连接

### MongoDB 连接字符串

```javascript
// 基本连接
mongodb://localhost:27017/mydb

// 带认证
mongodb://username:password@localhost:27017/mydb

// 副本集
mongodb://host1:27017,host2:27017,host3:27017/mydb?replicaSet=rs0

// 连接选项
mongodb://localhost:27017/mydb?maxPoolSize=10&minPoolSize=2&maxIdleTimeMS=30000
```

## 12.2 Mongoose 连接管理

### 基本连接

```typescript
import mongoose from 'mongoose';

// 连接 MongoDB
mongoose.connect('mongodb://localhost:27017/mydb', {
  maxPoolSize: 10,        // 最大连接数
  minPoolSize: 2,          // 最小连接数

  console.log('MongoDB 连接成功');

  console.error('MongoDB 连接错误:', err);
});

mongoose.connection.on('disconnected', () => {
  console.log('MongoDB 断开连接');

  process.exit(0);
```


### 连接池监控

```typescript
// 获取连接池状态
const poolSize = mongoose.connection.readyState;
const maxPoolSize = mongoose.connection.db?.serverConfig?.poolSize;

// 连接状态值
// 0 = disconnected（断开）
// 1 = connected（已连接）
// 2 = connecting（连接中）
// 3 = disconnecting（断开中）

// 监听连接池事件

  console.log('新连接建立:', connection.id);
});

mongoose.connection.on('disconnection', (connection) => {
  console.log('连接断开:', connection.id);

// 获取连接池统计信息
const getConnectionStats = () => {
  const db = mongoose.connection.db;
  if (!db) return null;
  
  const serverStatus = db.serverConfig?.s?.pool;
  return {
    totalConnections: serverStatus?.totalConnectionCount || 0,
    activeConnections: serverStatus?.availableConnectionCount || 0,
    waitQueueSize: serverStatus?.waitQueueSize || 0
  };
};

// 定期检查连接状态
setInterval(() => {
  const state = mongoose.connection.readyState;
  const stats = getConnectionStats();
  
  console.log('连接状态:', {
    0: '断开',
    1: '已连接',
    2: '连接中',
    3: '断开中'
  }[state]);
  
  if (stats) {
    console.log('连接池统计:', stats);
  }
}, 5000);
```

### 连接池单例模式

```typescript
// db/connection.ts
import { MongoClient, Db } from 'mongodb';

class Database {
  private client: MongoClient | null = null;
  private db: Db | null = null;

  async connect(uri: string, dbName: string): Promise<Db> {
    if (this.db) {
      return this.db;
    }

    this.client = new MongoClient(uri, {
      maxPoolSize: 10,
      minPoolSize: 2,
    });

    await this.client.connect();
    this.db = this.client.db(dbName);

    return this.db;
  }

  async disconnect(): Promise<void> {
    if (this.client) {
      await this.client.close();
      this.client = null;
      this.db = null;
    }
  }

  getDb(): Db {
    if (!this.db) {
      throw new Error('Database not connected');
    }
    return this.db;
  }
}

export const database = new Database();
```

## 12.5 性能优化策略

### 1. 连接池大小调优

```typescript
// 根据应用负载调整连接池大小
// 公式：maxPoolSize = (并发请求数 / 每个请求耗时) * 1.2

// 示例：100 并发，每个请求 100ms
// maxPoolSize = (100 / 0.1) * 1.2 = 1200
// 但实际建议：10-50 之间

const options = {
  maxPoolSize: 20,  // 根据实际情况调整
  minPoolSize: 5,  // 保持最小连接数
};
```

### 2. 查询优化

```typescript
// ❌ 差：不使用索引
const users = await User.find({ email: 'user@example.com' });

// ✅ 好：使用索引
await User.collection.createIndex({ email: 1 });
const users = await User.find({ email: 'user@example.com' });

// ❌ 差：返回所有字段
const users = await User.find();

// ✅ 好：只返回需要的字段
const users = await User.find().select('name email');

// ❌ 差：不使用 limit
const users = await User.find();

// ✅ 好：限制返回数量
const users = await User.find().limit(10);
```

### 3. 批量操作优化

```typescript
// ❌ 差：循环插入
for (const data of userData) {
  await User.create(data);
}

// ✅ 好：批量插入
await User.insertMany(userData);

// ❌ 差：循环更新
for (const user of users) {
  await User.updateOne({ _id: user._id }, { $set: { status: 'active' } });
}

// ✅ 好：批量更新
await User.updateMany(
  { _id: { $in: userIds } },
  { $set: { status: 'active' } }
);
```

### 4. 索引优化

```typescript
// 创建复合索引
await User.collection.createIndex({ email: 1, createdAt: -1 });

// 使用 explain 分析查询
const explain = await User.find({ email: 'user@example.com' }).explain();
console.log(explain.executionStats);

// 监控索引使用

  { $indexStats: {} }

```

### 5. 缓存策略

```typescript
import NodeCache from 'node-cache';

const cache = new NodeCache({ stdTTL: 600 }); // 10 分钟

async function getUserById(id: string) {
  // 先查缓存
  const cached = cache.get(id);
  if (cached) {
    return cached;
  }

  // 查数据库

  if (user) {
```

## 12.7 错误处理和重试

### 连接错误处理

```typescript
mongoose.connection.on('error', (err) => {
  console.error('MongoDB 连接错误:', err);
  
  // 根据错误类型处理
  if (err.name === 'MongoServerError') {
    // 服务器错误
    console.error('MongoDB 服务器错误');
  } else if (err.name === 'MongoNetworkError') {
    // 网络错误
    console.error('MongoDB 网络错误');
  }
});

// 重连逻辑
let reconnectAttempts = 0;
const maxReconnectAttempts = 5;

mongoose.connection.on('disconnected', () => {
  if (reconnectAttempts < maxReconnectAttempts) {
    reconnectAttempts++;
    setTimeout(() => {
      mongoose.connect(uri).catch(console.error);
    }, 5000 * reconnectAttempts);
  }
});
```

### 查询重试

```typescript
async function retryQuery<T>(
  queryFn: () => Promise<T>,
  maxRetries: number = 3,
  delay: number = 1000
): Promise<T> {
  for (let i = 0; i < maxRetries; i++) {
    try {
      return await queryFn();
    } catch (error) {
      if (i === maxRetries - 1) {
        throw error;
      }
      await new Promise(resolve => setTimeout(resolve, delay * (i + 1)));
    }
  }
  throw new Error('Max retries exceeded');
}

// 使用
const user = await retryQuery(() => User.findById(userId));
```
