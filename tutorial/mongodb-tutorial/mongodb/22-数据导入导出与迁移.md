# 第二十二章：数据导入导出与迁移

## 22.1 数据导出（mongoexport）

### 基本导出

```bash
# 导出整个集合为 JSON
mongoexport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --out=users.json

# 导出为 CSV
mongoexport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --type=csv \
  --fields=name,email,age \
  --out=users.csv

# 带查询条件导出
mongoexport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --query='{"age":{"$gte":18}}' \
  --out=adults.json

# 导出特定字段
mongoexport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --fields=name,email \
  --out=users.json
```

### 高级导出选项

```bash
# 带认证导出
mongoexport --uri="mongodb://username:password@localhost:27017/mydb" \
  --collection=users \
  --out=users.json

# 从副本集导出
mongoexport --uri="mongodb://localhost:27017/mydb?replicaSet=rs0" \
  --collection=users \
  --out=users.json

# 限制导出数量
mongoexport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --limit=1000 \
  --out=users.json

# 跳过文档
mongoexport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --skip=100 \
  --limit=100 \
  --out=users.json
```

## 22.2 数据导入（mongoimport）

### 基本导入

```bash
# 从 JSON 导入
mongoimport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --file=users.json

# 从 CSV 导入
mongoimport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --type=csv \
  --headerline \
  --file=users.csv

# 导入时指定字段
mongoimport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --type=csv \
  --fields=name,email,age \
  --file=users.csv
```

### 导入选项

```bash
# 更新已存在文档（upsert）
mongoimport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --file=users.json \
  --upsert \
  --upsertFields=email

# 删除集合后导入
mongoimport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --file=users.json \
  --drop

# 忽略错误继续导入
mongoimport --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --file=users.json \
  --ignoreBlanks \
  --numInsertionWorkers=4
```

## 22.3 数据备份（mongodump）

### 基本备份

```bash
# 备份整个数据库
mongodump --uri="mongodb://localhost:27017/mydb" \
  --out=/backup/mydb

# 备份特定集合
mongodump --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --out=/backup/users

# 压缩备份
mongodump --uri="mongodb://localhost:27017/mydb" \
  --out=/backup/mydb \
  --gzip

# 备份到归档文件
mongodump --uri="mongodb://localhost:27017/mydb" \
  --archive=/backup/mydb.archive
```

### 高级备份选项

```bash
# 备份时查询
mongodump --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  --query='{"age":{"$gte":18}}' \
  --out=/backup/adults

# 备份索引
mongodump --uri="mongodb://localhost:27017/mydb" \
  --out=/backup/mydb \
  --excludeCollection=logs  # 排除集合

# 从副本集备份
mongodump --uri="mongodb://localhost:27017/mydb?replicaSet=rs0" \
  --readPreference=secondary \
  --out=/backup/mydb
```

## 22.4 数据恢复（mongorestore）

### 基本恢复

```bash
# 恢复整个数据库
mongorestore --uri="mongodb://localhost:27017/mydb" \
  /backup/mydb

# 恢复特定集合
mongorestore --uri="mongodb://localhost:27017/mydb" \
  --collection=users \
  /backup/mydb/users.bson

# 从压缩文件恢复
mongorestore --uri="mongodb://localhost:27017/mydb" \
  --gzip \
  /backup/mydb

# 从归档文件恢复
mongorestore --uri="mongodb://localhost:27017/mydb" \
  --archive=/backup/mydb.archive
```

### 恢复选项

```bash
# 恢复时删除现有数据
mongorestore --uri="mongodb://localhost:27017/mydb" \
  /backup/mydb \
  --drop

# 恢复索引
mongorestore --uri="mongodb://localhost:27017/mydb" \
  /backup/mydb \
  --noIndexRestore  # 不恢复索引

# 恢复到不同数据库
mongorestore --uri="mongodb://localhost:27017" \
  --nsFrom="mydb.users" \
  --nsTo="newdb.users" \
  /backup/mydb
```

## 22.5 数据迁移策略

### 策略 1：全量迁移

```bash
# 1. 停止应用写入
# 2. 执行备份
mongodump --uri="mongodb://source:27017/mydb" \
  --out=/backup/mydb

# 3. 恢复数据
mongorestore --uri="mongodb://target:27017/mydb" \
  /backup/mydb

# 4. 验证数据
# 5. 切换应用连接
```

### 策略 2：增量迁移

```bash
# 1. 初始全量备份
mongodump --uri="mongodb://source:27017/mydb" \
  --out=/backup/initial

# 2. 恢复初始数据
mongorestore --uri="mongodb://target:27017/mydb" \
  /backup/initial

# 3. 使用变更流同步增量数据
# （见变更流章节）

# 4. 切换应用连接
```

### 策略 3：在线迁移

```typescript
// 使用变更流实时同步
async function migrateData() {
  const sourceClient = new MongoClient(sourceUri);
  const targetClient = new MongoClient(targetUri);
  
  await sourceClient.connect();
  await targetClient.connect();
  
  const sourceDb = sourceClient.db('mydb');
  const targetDb = targetClient.db('mydb');
  
  // 初始同步
  const sourceData = await sourceDb.collection('users').find().toArray();
  await targetDb.collection('users').insertMany(sourceData);
  
  // 监听变更
  const changeStream = sourceDb.collection('users').watch();
  changeStream.on('change', async (change) => {
    // 同步变更到目标数据库
    await syncChange(targetDb, change);
  });
}
```

## 22.6 跨数据库迁移

### MongoDB 到 MongoDB

```bash
# 方法 1：使用 mongodump/mongorestore
mongodump --uri="mongodb://source:27017/sourcedb" \
  --out=/backup
mongorestore --uri="mongodb://target:27017/targetdb" \
  /backup/sourcedb

# 方法 2：使用 mongoexport/mongoimport
mongoexport --uri="mongodb://source:27017/sourcedb" \
  --collection=users \
  --out=users.json
mongoimport --uri="mongodb://target:27017/targetdb" \
  --collection=users \
  --file=users.json
```

### 从 MySQL 迁移

```typescript
import mysql from 'mysql2/promise';
import { MongoClient } from 'mongodb';

async function migrateFromMySQL() {
  const mysqlConn = await mysql.createConnection({
    host: 'localhost',
    user: 'root',
    password: 'password',
    database: 'mydb'
  });
  
  const mongoClient = new MongoClient('mongodb://localhost:27017');
  await mongoClient.connect();
  const mongoDb = mongoClient.db('mydb');
  
  // 查询 MySQL 数据
  const [rows] = await mysqlConn.execute('SELECT * FROM users');
  
  // 插入 MongoDB
  await mongoDb.collection('users').insertMany(rows);
  
  await mysqlConn.end();
  await mongoClient.close();
}
```

### 从 PostgreSQL 迁移

```typescript
import { Pool } from 'pg';
import { MongoClient } from 'mongodb';

async function migrateFromPostgreSQL() {
  const pgPool = new Pool({
    host: 'localhost',
    user: 'postgres',
    password: 'password',
    database: 'mydb'
  });
  
  const mongoClient = new MongoClient('mongodb://localhost:27017');
  await mongoClient.connect();
  const mongoDb = mongoClient.db('mydb');
  
  const result = await pgPool.query('SELECT * FROM users');
  
  await mongoDb.collection('users').insertMany(result.rows);
  
  await pgPool.end();
  await mongoClient.close();
}
```

## 22.8 最佳实践

### 1. 备份前验证

```bash
# ✅ 好：备份前检查连接
mongosh --uri="mongodb://localhost:27017/mydb" --eval "db.stats()"

# ✅ 好：检查磁盘空间
df -h  # Linux
Get-PSDrive C  # Windows
```

### 2. 增量备份

```bash
# ✅ 好：使用 oplog 进行增量备份
mongodump --uri="mongodb://localhost:27017/mydb" \
  --oplog \
  --out=/backup/incremental
```

### 3. 压缩备份

```bash
# ✅ 好：使用压缩减少存储空间
mongodump --uri="mongodb://localhost:27017/mydb" \
  --gzip \
  --out=/backup/mydb
```

### 4. 定期备份

```bash
# ✅ 好：使用 cron 定时备份
# 每天凌晨 2 点备份
0 2 * * * mongodump --uri="mongodb://localhost:27017/mydb" --gzip --out=/backup/$(date +\%Y\%m\%d)
```
